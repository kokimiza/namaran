LockFile.hs

import Control.Exception (Exception, catch, throwIO)

data BadFile = BadFile String deriving (Show)

instance Exception BadFile

validateContents :: String -> IO ()
validateContents name =
  if name == "corrupt.txt"
    then throwIO (BadFile "bad contents")
    else putStrLn ("process " ++ name)

processFile :: String -> IO ()
processFile name = do
  putStrLn ("open " ++ name)
  validateContents name
  putStrLn ("close " ++ name)

main :: IO ()
main =
  processFile "corrupt.txt"
    `catch` \(BadFile msg) -> putStrLn ("error: " ++ msg)

processFileは、ファイルを開き、中身を検証し、(検証が成功しても失敗しても)必ず閉じるつもりです。processFile "corrupt.txt"を実行すると、いまの出力はopen corrupt.txtとerror: bad contentsの2行だけです。本来はopen corrupt.txt・close corrupt.txt・error: bad contentsの3行が出るべきです(中身が不正だと分かっても、開いたファイルは閉じるべきなので)。何が問題で、どう直せばよいでしょうか?

Answer

processFileのdoブロックは、putStrLn ("open " ++ name)・validateContents name・putStrLn ("close " ++ name)という3つの文を上から順に実行するだけの、逐次実行として書かれています。validateContentsが例外を投げると、C++のデストラクタやRustのDropのようにスコープを抜けるときに自動実行される仕組みがHaskellには無いため、doブロックの残りの文(closeを出力する行)はそのまま読み飛ばされ、例外は直接mainのcatchまで伝わってしまいます。「後始末はdoブロックの最後に置けば必ず実行される」という思い込みが、この見落としの原因です。Control.Exceptionのfinallyを使い、「開く・検証する」の部分と「閉じる」の部分を明示的に分けて、後者を必ず実行されるようにします: 

processFile :: String -> IO ()
processFile name =
  ( do
      putStrLn ("open " ++ name)
      validateContents name
  )
    `finally` putStrLn ("close " ++ name)
