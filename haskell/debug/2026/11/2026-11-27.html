data Re = Chr Char | Opt Re | Star Re

match
  :: Re -> String -> (String -> Bool) -> Bool
match (Chr c) (x : xs) k = x == c && k xs
match (Chr _) [] _ = False
match (Opt r) s k = match r s k || k s
match (Star r) s k = k s || match r s next
 where
  next t = match (Star r) t k

main :: IO ()
main = do
  let re = Star (Opt (Chr 'a'))
  print
    [match re s (== "b") | s <- ["ab", "a"]]

match re s k は、正規表現 re が s の先頭に一致する食べ方のどれかについて、食べ残しを k に渡して True になれば True を返すつもりの関数です。Chr c は文字 c、Opt r は r?、Star r は r* に当たります。main は (a?)* のあとに b が1つだけ残るかを、"ab" と "a" について出力します。

match (Star r) s k = k s || match r s next
 where
  next t = match (Star r) t k

"ab" では照合が終わります。Star は k s を先に試し、だめなら r を1回使ってから next で自分に戻ります。a? が a を1回食べたところで k "b" が True になり、|| は右側を評価せずに True を返します。食べ方を最後まで調べる前に答えが出るので、欠陥は表に出ません。

"a" にはどの食べ方でも b が残らないので、すべての食べ方を調べ切らなければなりません。a を1つ食べて s が "" になると、Opt の match r s k は失敗し、右側の k s に回ります。ここでの k は next なので、a? が何も食べないまま match (Star r) "" k がもう一度呼ばれます。引数は前回とまったく同じなので、同じ道を何度でもたどり、False を返す時が来ません。1つ目の True は求まっていますが、リストの表示が改行まで進まないので、画面には何も出ません。

Right 時間計算量 𝒪(n2) 空間計算量 𝒪(n) n は main の re で照合する文字列の長さ 

match (Star r) s k = k s || match r s next
 where
  next t = t /= s && match (Star r) t k

r が1文字も食べなかったとき、つまり食べ残し t が s と同じときは、Star に戻らずにその道を捨てます。何も食べない場合はすでに先頭の k s で試しているので、捨てても一致を見逃しません。Star に戻るたびに文字列は1文字以上短くなるので、どんな入力でも再帰は文字列の長さの分で止まり、"a" は False になります。t /= s は等しい2つの文字列を最後まで比べるので1回に n に比例する手間がかかり、全体では n の2乗に比例します。
