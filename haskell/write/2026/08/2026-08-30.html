Parse.hs

class Parse a where
  parse :: String -> Maybe a

instance Parse Bool where
  parse "True" = Just True
  parse "False" = Just False
  parse _ = Nothing

instance Parse Int where
  parse s = case reads s of
    [(n, "")] -> Just n
    _ -> Nothing

instance (Parse a, Parse b) => Parse (a, b) where

parse は文字列を型 a の値として読み、読めなければ Nothing を返します。Parse a と Parse b のインスタンスがあるとき、"左,右" の形の文字列を組 (a, b) として読む Parse (a, b) のインスタンスを書いてください。最初のコンマの左を a、右の残り全部を b として読み、どちらかが読めないか、コンマが無ければ Nothing です。

入力例 1

parse "12,True" :: Maybe (Int, Bool)

出力例 1

Just (12,True)

入力例 2

parse "1,2,False" :: Maybe (Int, (Int, Bool))

出力例 2

Just (1,(2,False))

入力例 3

parse "12" :: Maybe (Int, Bool)

出力例 3

Nothing

instance (Parse a, Parse b) => Parse (a, b) where
  parse s = case break (== ',') s of
    (l, ',' : r) -> do
      x <- parse l
      y <- parse r
      return (x, y)
    _ -> Nothing

break (== ',') s は、s を最初のコンマの手前で2つに分けます。コンマがあれば右側はコンマから始まるので、パターン (l, ',' : r) でコンマを外した残りを r に取り出します。コンマが無ければ右側は空になり、2つ目の腕で Nothing を返します。parse l と parse r は同じ名前の関数を呼んでいるように見えますが、使われるインスタンスは別です。組み立てる結果の型が Maybe (a, b) なので、parse l の型は Maybe a、parse r の型は Maybe b に決まり、インスタンスの文脈 (Parse a, Parse b) で受け取った a と b それぞれの parse が呼ばれます。引数の型ではなく、返す値の型でどの実装を使うかが決まります。Maybe の do は、parse l か parse r のどちらかが Nothing ならそこで Nothing になり、両方が Just のときだけ return (x, y) で組を Just に包みます。

入力例 2 の型 (Int, (Int, Bool)) では、右側の b が組 (Int, Bool) です。最初のコンマで "1" と "2,False" に分けると、parse "2,False" には今書いた Parse (a, b) のインスタンスがもう一度使われ、Just (2,False) が返ります。入れ子の組のための特別な処理は要りません。最後のコンマで分けると、左の "1,2" を Int として読めずに Nothing になるので、区切るのは最初のコンマです。入力例 3 はコンマが無いので Nothing です。掲載したのは参考実装で、唯一の正解ではありません。do の代わりに、case (parse l, parse r) of で場合分けして (Just x, Just y) -> Just (x, y) と書いてもかまいません。
