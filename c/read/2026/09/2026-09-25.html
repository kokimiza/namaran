turnstile.c

#include <stdio.h>

enum state { LOCKED, UNLOCKED };

int main(void) {
    const char *events = "cppccp";
    enum state st = LOCKED;
    int passed = 0, blocked = 0;

    for (const char *e = events; *e != '\0'; e++) {
        switch (st) {
        case LOCKED:
            if (*e == 'c') {
                st = UNLOCKED;
            } else {
                blocked++;
            }
            break;
        case UNLOCKED:
            if (*e == 'p') {
                passed++;
                st = LOCKED;
            }
            break;
        }
    }
    printf("passed=%d blocked=%d\n", passed, blocked);
    return 0;
}

これは何を出力するでしょうか?

Answer

passed=2 blocked=1 です。c(硬貨)でLOCKED→UNLOCKED、p(押す)で通過してpassedが1、LOCKEDに戻ります。次のpはLOCKEDのときに来たのでblockedが1。続くcでUNLOCKEDになり、その次のcはUNLOCKEDの分岐で何にも当たらず、状態は変わりません(2枚目の硬貨は通過を2回にしない)。最後のpでpassedが2になります。外側のswitchで「いまの状態」を、内側のifで「来た入力」を分けるのが、Cで状態機械を書くときの基本の形です。同じpでも、どのcaseの中で読まれたかで意味が変わります。
