job_limiter.c

#define _POSIX_C_SOURCE 200809L
#include <semaphore.h>
#include <stdio.h>

int main(void) {
    sem_t sem;
    sem_init(&sem, 0, 2);

    for (int i = 0; i < 3; i++) {
        if (sem_trywait(&sem) == 0) {
            printf("job %d: acquired\n", i);
        } else {
            printf("job %d: blocked\n", i);
        }
    }

    sem_post(&sem);
    printf("job 3: %s\n", sem_trywait(&sem) == 0 ? "acquired" : "blocked");

    sem_destroy(&sem);
}

semは同時に2つまで許可を出す(sem_initの第3引数)カウンティングセマフォです。sem_trywaitはブロックせずに許可を1つ取ろうとし、取れれば0、取れなければ-1を返します。このコードは何を出力するでしょうか?

Answer

job 0: acquired、job 1: acquired、job 2: blocked、job 3: acquired(各行の後に改行)。semは初期値2で作られているので、最初の2回のsem_trywait(job 0とjob 1)はどちらも許可を取れて値を0まで減らします。3回目(job 2)の時点で残りの許可は0なので、sem_trywaitはブロックする代わりに即座に失敗を返し、job 2はblockedになります。ループの後のsem_postが許可を1つ返すことでセマフォの値が1に戻り、続くjob 3のsem_trywaitはその1つを取れるのでacquiredになります。これが「同時に実行できる仕事の数をNに制限する」というセマフォの基本的な使い方で、Nを超えて許可を要求した側は、誰かが許可を返すまで先に進めません。
