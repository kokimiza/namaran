try_run_job.c

#include <semaphore.h>

int try_run_job(sem_t *limiter) {
    // write this: try to take one permit from limiter without blocking;
    // return 1 (and hold the permit) if one was available, 0 otherwise
}

limiterは同時実行数を制限するために使うカウンティングセマフォです。try_run_jobは、ブロックせずに許可を1つ取れれば1を返し(そのまま許可を保持する)、上限に達していて取れなければ0を返してください。limiterの初期値が1のとき、1回目の呼び出しは1を返します。直後の2回目の呼び出しは、まだ誰も許可を返していないので0を返します。その後どこかで許可が1つ返却されてから3回目を呼ぶと、再び1が返ります。

int try_run_job(sem_t *limiter) {
    return sem_trywait(limiter) == 0;
}

sem_trywaitはブロックせずに許可を1つ取ろうとする関数で、取れれば0、取れなければ-1を返します(errnoにEAGAINを設定します)。求められているのは「取れたかどうか」という真偽値なので、戻り値を0と比較した結果(1または0)をそのまま返せば十分です。許可を実際に取れた場合、その許可の返却(sem_post)は呼び出し側の責任になります。これは参考実装であり、唯一の正解ではありません。
