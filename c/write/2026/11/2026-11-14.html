print_reading.c

struct reading {
    int sensor;
    char grade;
    unsigned char flags;
    double value;
    const char* label;
};

void print_reading(struct reading r) {
}

センサーの読み取り値 r のフィールドを、宣言の順に空白1つで区切って1行に出す print_reading を書いてください。char のフィールドは文字として、signed char・unsigned char・int は10進の整数として、double は printf の %g で、文字列はそのまま出し、行の最後に改行を出します。フィールドの型は、あとで char、signed char、unsigned char、int、double、const char*、char* のどれかに変わることがあります。そのときも print_reading の本文は直さずに済むようにしてください。

入力例 1

print_reading((struct reading){7, 'B', 200, 21.5, "east"})

出力例 1

7 B 200 21.5 east

入力例 2

print_reading((struct reading){-3, 'a', 255, 1e-5, "n/a"})

出力例 2

-3 a 255 1e-05 n/a

void print_reading(struct reading r) {
#define FMT(v)               \
    _Generic((v),            \
        char: "%c",          \
        signed char: "%d",   \
        unsigned char: "%d", \
        int: "%d",           \
        double: "%g",        \
        const char*: "%s",   \
        char*: "%s")
#define PUT(v, end) (printf(FMT(v), (v)), putchar(end))
    PUT(r.sensor, ' ');
    PUT(r.grade, ' ');
    PUT(r.flags, ' ');
    PUT(r.value, ' ');
    PUT(r.label, '\n');
#undef PUT
#undef FMT
}

型が変わっても本文を直さずに済ませるには、書式をフィールドの型から選ばせます。参考実装の FMT(v) は、_Generic で v の型に合う書式の文字列リテラルを選びます。PUT(v, end) はその書式で v を出し、続けて区切りの文字を出します。どちらも関数の中だけで使うマクロなので、最後に #undef で消しています。掲載したのは参考実装で、唯一の正解ではありません。

関連付けに char、signed char、unsigned char の3つを並べているのは、この3つが互いに別の型だからです。char の表現が符号付きと同じ処理系でも、char は signed char の別名ではありません。char だけを書くと flags の unsigned char はどの関連付けにも当たらず、default も無いのでコンパイルエラーになります。逆に char を %d の側にまとめると、'B' が 66 と出てしまいます。

文字列も同じで、const char* と char* は指す先の修飾が違う別の型です。いまの label は const char* ですが、char* に変わっても当たるように両方を並べます。signed char と unsigned char を %d で出せるのは、可変長引数に渡すときに int へ昇格するからです。入力例1では、flags の 200 が int の 200 として渡り、%d でそのまま出ます。
