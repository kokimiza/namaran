sum_digits.c

#include <setjmp.h>
#include <stdio.h>

static jmp_buf on_error;

static int digit(char c) {
    if (c < '0' || c > '9') {
        longjmp(on_error, 1);
    }
    return c - '0';
}

int sum_digits(const char* s) {
}

文字列 s の各文字を digit で数に直し、その和を返す sum_digits を書いてください。digit は数字でない文字を受け取ると longjmp(on_error, 1) を呼んで戻ってきません。そのときは sum_digits が -1 を返すようにします。空の文字列なら 0 です。

入力例 1

sum_digits("2026")

出力例 1

10

入力例 2

sum_digits("12a4")

出力例 2

-1

入力例 3

sum_digits("")

出力例 3

0

#include <setjmp.h>
#include <stdio.h>

static jmp_buf on_error;

static int digit(char c) {
    if (c < '0' || c > '9') {
        longjmp(on_error, 1);
    }
    return c - '0';
}

int sum_digits(const char* s) {
    if (setjmp(on_error) != 0) {
        return -1;
    }
    int sum = 0;
    for (; *s != '\0'; s++) {
        sum += digit(*s);
    }
    return sum;
}

longjmp は、on_error に記録した setjmp の呼び出し位置へ戻ります。戻り先の関数は、longjmp が呼ばれた時点でまだ実行中でなければなりません。そこで setjmp は sum_digits の中で、digit を呼ぶ前に呼びます。初めて呼ばれた setjmp は 0 を返すので、そのまま和を数える処理へ進みます。longjmp(on_error, 1) で戻ってきたときは 1 を返すので、if の中で -1 を返します。掲載したのは参考実装で、唯一の正解ではありません。

入力例 2 では、'1' と '2' を足したあと、'a' を受け取った digit が longjmp を呼びます。制御は sum_digits の先頭の setjmp に戻り、ループの続きの '4' は読まれずに -1 が返ります。入力例 3 の空の文字列ではループが一度も回らず、sum の初期値 0 が返ります。

setjmp を呼ぶだけの補助関数を作り、それを sum_digits から呼ぶ書き方では動きません。補助関数が戻った時点でその呼び出しは終わっているので、そこへ longjmp で戻ることはできず、未定義動作になります。
