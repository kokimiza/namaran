escape_bounds.c

#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

size_t escape(const char *src, char *dst) {
    size_t j = 0;
    for (size_t i = 0; src[i] != '\0'; i++) {
        if (src[i] == '"' || src[i] == '\\') {
            dst[j++] = '\\';
        }
        dst[j++] = src[i];
    }
    dst[j] = '\0';
    return j;
}

int main(void) {
    const char *src = "\"\"\"\"";
    char *dst = malloc(strlen(src) + 1);
    size_t n = escape(src, dst);
    printf("%s (%zu)\n", dst, n);
    free(dst);
    return 0;
}

エスケープした文字列をヒープに確保したバッファへ書き込み、表示するプログラムです。入力は二重引用符4つからなる文字列(実行時の中身は"""")。escapeは"や\\を見つけるたびに、その1文字を\\+その文字の2文字へ展開しますが、呼び出し側はmalloc(strlen(src) + 1)で入力と同じ長さぶんしかバッファを確保していません。いまの出力は保証されません(ヒープ領域の境界を超えて書き込むため、AddressSanitizerがheap-buffer-overflowを検出することもあれば、たまたま動いてしまうこともあります)。本来の出力は\"\"\"\" (8)です。何が問題で、どう直せばよいでしょうか?

Answer

エスケープ後の文字列は、入力中の特殊文字("と\\)1つにつき1文字ずつ長くなります。最悪の場合(すべての文字が特殊文字のとき)、出力は入力の2倍の長さになり得るので、確保すべきバッファサイズはstrlen(src) + 1ではなく、終端の'\0'ぶんも含めた2 * strlen(src) + 1です。ここでは入力4文字すべてがダブルクォートなので、エスケープ後は8文字+終端1バイトの計9バイトが必要ですが、確保されているのは5バイトだけで、4バイトぶん範囲外へ書き込んでしまいます。直し方は確保サイズを2倍に直すだけです: 

char *dst = malloc(2 * strlen(src) + 1);
