count_lines.c

#include <stdio.h>

static size_t count_lines(FILE *f) {
    char buf[8];
    size_t lines = 0;
    while (!feof(f)) {
        fread(buf, 1, sizeof buf, f);
        for (size_t i = 0; i < sizeof buf; i++)
            if (buf[i] == '\n') lines++;
    }
    return lines;
}

int main(void) {
    FILE *f = tmpfile();
    if (!f) return 1;
    fputs("one\ntwo\nthree\n", f);
    rewind(f);
    printf("%zu\n", count_lines(f));
}

count_lines は、どれほど大きなファイルでも8バイトのバッファだけで読み進め、改行を数えて行数を返すつもりの関数です。main は one\ntwo\nthree\n を書いた一時ファイルを渡すので、引数も標準入力も使いません。いまは 4 が出ますが、本来は 3 が出るはずです。

Answer

Wrong

size_t lines = 0;
while (!feof(f)) {
    fread(buf, 1, sizeof buf, f);
    for (size_t i = 0; i < sizeof buf; i++)
        if (buf[i] == '\n') lines++;
}

ファイルは14バイトあります。1回目の fread は one\ntwo\n の8バイトで buf を埋め、改行を2つ数えます。2回目に残っているのは three\n の6バイトだけなので、fread が書き換えるのは buf[0] から buf[5] までで、buf[6] と buf[7] には1回目の two\n の末尾の o\n が残ります。ところが for は sizeof buf の8バイトをすべて調べるので、今回読んだ \n に加えて、前回の残りの \n も数えます。合計は4です。

このファイルでは2回目の fread が終わりに当たるので、feof が真になり、ループはそこで抜けます。ただし feof が真になるのは、読もうとして終わりに当たったあとです。ファイルがちょうど16バイトなら2回目の fread は8バイトを読み切ってまだ終わりに当たらず、3回目は1バイトも読めないまま、前回の8バイトを丸ごともう一度数えます。

Right

size_t lines = 0, n;
while ((n = fread(buf, 1, sizeof buf, f)) > 0) {
    for (size_t i = 0; i < n; i++)
        if (buf[i] == '\n') lines++;
}

fread は実際に読めたバイト数を返すので、それを n に受け取り、buf の先頭から n バイトだけを調べます。前回の残りは見なくなり、2回目は three\n の改行1つだけを数えて 3 になります。ループの条件も fread の戻り値にしたので、1バイトも読めなかった時点で抜け、古い中身を数え直すこともなくなります1。

  1. while (!feof(f)) は、読み取りエラーのときにも抜けられません。エラーでは feof ではなく ferror が真になり、fread は0を返し続けるので、ループが止まらなくなります。
