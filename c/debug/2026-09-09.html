fetch_in_batches.c

#include <stdio.h>

#define BATCH 3

static void fetch_users_in(const int *ids, size_t count) {
    printf("SELECT * FROM users WHERE id IN (");
    for (size_t i = 0; i < count; i++) {
        printf("%s%d", i > 0 ? ", " : "", ids[i]);
    }
    printf(")\n");
}

static void fetch_all(const int *ids, size_t n) {
    size_t batches = n / BATCH;
    for (size_t b = 0; b < batches; b++) {
        size_t start = b * BATCH;
        size_t count = n - start < BATCH ? n - start : BATCH;
        fetch_users_in(ids + start, count);
    }
}

int main(void) {
    int ids[] = {11, 12, 13, 14, 15, 16, 17};
    fetch_all(ids, sizeof ids / sizeof ids[0]);
    return 0;
}

fetch_all は、N+1クエリを避けるために、ユーザーIDを BATCH 件ずつの IN 句にまとめて取得するつもりの関数です。入力は main のとおり、7件のID {11, 12, 13, 14, 15, 16, 17} です。いまの出力は SELECT * FROM users WHERE id IN (11, 12, 13) と SELECT * FROM users WHERE id IN (14, 15, 16) の2行だけです。本来は、この2行に続けて3行目に SELECT * FROM users WHERE id IN (17) が出力されます。どこを直せばよいでしょうか?

Answer

size_t batches = n / BATCH; を size_t batches = (n + BATCH - 1) / BATCH; に直します。

整数どうしの / は小数部を切り捨てるので、7 / 3 は2です。ループは2周で終わり、端数の 17 は送られません。count の計算は3件に満たない最後のバッチも正しく扱えますが、その周回が来ません。

割る前に BATCH - 1 を足すと、割り切れないときだけ商が1つ増えます。7件なら 9 / 3 で3バッチ、6件なら 8 / 3 で2バッチです。ループを for (size_t start = 0; start < n; start += BATCH) のように先頭の位置で回せば、バッチ数を求める式そのものが要らなくなります。
