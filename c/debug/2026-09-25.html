sum_line.c

#include <ctype.h>
#include <stdio.h>

enum state { BETWEEN, NUMBER };

int sum_line(const char *s) {
    enum state st = BETWEEN;
    int sum = 0, cur = 0;
    for (const char *p = s; *p != '\0'; p++) {
        switch (st) {
        case BETWEEN:
            if (*p == '#') {
                break;
            }
            if (isdigit((unsigned char)*p)) {
                cur = *p - '0';
                st = NUMBER;
            }
            break;
        case NUMBER:
            if (isdigit((unsigned char)*p)) {
                cur = cur * 10 + (*p - '0');
            } else {
                sum += cur;
                st = BETWEEN;
            }
            break;
        }
    }
    if (st == NUMBER) {
        sum += cur;
    }
    return sum;
}

int main(void) {
    printf("%d\n", sum_line("10 20 # 30"));
    return 0;
}

sum_lineは、1行に並んだ数値を合計し、#から後ろはコメントとして無視するつもりの関数です。入力はsum_line("10 20 # 30")。いまの出力は60ですが、本来の出力は30です。どこを直せばよいでしょうか?

Answer

case BETWEEN:の中のif (*p == '#') { break; }が抜けるのは、いちばん内側のswitchだけです。forループはそのまま次の文字へ進むので、#は「数字でない文字」として読み飛ばされ、後ろの30も合計に入ります。breakは、それを囲む最も内側のswitchか繰り返し文に結びつく、という1つの規則です。状態機械をfor+switchで書くと、「ループを終えたい」breakと「このcaseを終えたい」breakが同じ場所に並ぶので、取り違えが起きやすくなります(1990年のAT&T長距離電話網の大規模障害も、switchの中のbreakの取り違えが原因だったとされます)。BETWEENの状態では読みかけの数値はすでにsumに足されているので、最小の修正はそのbreak;をreturn sum;にすることです。状態を増やしたい場合は、DONE状態を足してループ条件で見る方法もあります。
