run_job.c

#define _POSIX_C_SOURCE 200809L
#include <semaphore.h>
#include <stdio.h>

sem_t limiter;

void run_job(int id, int fail) {
    sem_wait(&limiter);
    if (fail) {
        printf("job %d: error, aborting\n", id);
        return;
    }
    printf("job %d: done\n", id);
    sem_post(&limiter);
}

int main(void) {
    sem_init(&limiter, 0, 1);
    run_job(0, 1);
    printf("job 1: %s\n", sem_trywait(&limiter) == 0 ? "admitted" : "rejected");
    sem_destroy(&limiter);
}

run_jobはlimiterから許可を1つ取り、仕事が終わったら(成功でも失敗でも)必ずその許可を返すつもりです。run_job(0, 1)を呼んで(エラーで失敗させる)からrun_job(1, 0)相当の許可取得を試みる(コード中ではsem_trywaitで直接確認)と、いまの出力はjob 0: error, abortingの後にjob 1: rejectedになります。本来はlimiterの初期値が1で、job 0が(成功でも失敗でも)許可を返し終えているはずなので、job 1: admittedになるべきです。何が問題で、どう直せばよいでしょうか?

Answer

run_jobはfailが真のとき、エラーメッセージを出してすぐreturnしていますが、その手前で呼んだsem_waitに対応するsem_postを呼ばずに抜けています。正常終了パスだけが許可を返し、エラーパスは許可を返さないまま関数を抜けてしまうので、その許可は誰にも返却されないまま失われます(リーク)。同時に実行できる仕事の数を制限するはずのセマフォが、エラーが1回起きるたびに上限を1つずつ静かに減らしていく形になり、いずれ本来なら実行できるはずの仕事までrejectedになってしまいます。修正は、エラーパスでもreturnする前にsem_postを呼び、取った許可を必ず返すことです: 

void run_job(int id, int fail) {
    sem_wait(&limiter);
    if (fail) {
        printf("job %d: error, aborting\n", id);
        sem_post(&limiter);
        return;
    }
    printf("job %d: done\n", id);
    sem_post(&limiter);
}
