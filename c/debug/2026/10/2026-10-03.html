bracket.c

#include <stdio.h>
#include <stdlib.h>

static size_t utf8_len(const char *s) {
    size_t n = 0;
    for (; *s; s++)
        n += (*s & 0xC0) != 0x80;
    return n;
}

int main(void) {
    const char *s = "東京";
    char *out = malloc(utf8_len(s) + 3);
    if (!out)
        return 1;
    sprintf(out, "[%s]", s);
    puts(out);
    free(out);
}

このプログラムは、文字列 s を角括弧で囲んだ文字列を新しく確保した領域に作り、出力します。引数も標準入力も使わず、s は "東京" です。いまは [東京] と出ることもありますが、AddressSanitizer を付けてビルドすると heap-buffer-overflow を報告して止まります。本来は、どうビルドしても [東京] と出て正常に終わるはずです。

Answer

Wrong

char *out = malloc(utf8_len(s) + 3);
if (!out)
    return 1;
sprintf(out, "[%s]", s);

utf8_len は継続バイト(10xxxxxx)を数えないので、返すのはバイト数ではなく文字数です。"東京" は2文字なので、malloc が確保するのは 2 + 3 で5バイトです。一方、sprintf が書き込むのはバイト列で、東 と 京 は UTF-8 でそれぞれ3バイトあります。角括弧2つと終端の '\0' を合わせた9バイトが、5バイトの領域に書き込まれます。

はみ出した4バイトは確保していないヒープの領域で、そこへの書き込みは未定義動作です。malloc は頼んだ大きさより少し大きい塊を返すことが多いので、何事もなく [東京] と出る場合もあります。AddressSanitizer は確保した大きさちょうどの外側を見張っているので、この書き込みを見つけて止めます。

Right

#include <string.h>

char *out = malloc(strlen(s) + 3);

<string.h> を読み込み、strlen で大きさを決めます。strlen は終端の手前までのバイト数を返すので、"東京" なら 6 です。[、6バイトの本体、]、'\0' の合計9バイトと、確保する大きさが一致します。確保する大きさは、そこへ書き込む sprintf と同じバイト単位で数える必要があります1。

  1. snprintf(NULL, 0, "[%s]", s) は何も書き込まずに、書き込むはずのバイト数(終端を除く)を返します。これに1を足して確保すれば、書式を変えても大きさがずれません。
