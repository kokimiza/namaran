has_opt.c

#include <stdio.h>
#include <string.h>

bool has_opt(char* argv[], const char* opt) {
    for (int i = 1; argv[i]; i++) {
        if (strcmp(argv[i], opt)) {
            return true;
        }
    }
    return false;
}

int main(void) {
    char* v[] = {"prog", "-q", NULL};
    printf("%d\n", has_opt(v, "-q"));
    printf("%d\n", has_opt(v, "-x"));
}

has_opt は、argv の argv[1] 以降に opt と同じ文字列があるかを返すつもりの関数です。main はコマンドライン prog -q に当たる配列を作り、-q と -x があるかを調べます。本来は 1 と 0 を1行ずつ出力するはずが、いまは 0 と 1 が出力されます。

Answer

Wrong

if (strcmp(argv[i], opt)) {
    return true;
}

strcmp は2つの文字列を比べ、等しければ 0 を、等しくなければ、どちらが辞書順で先かに応じて負か正の値を返します。if の条件は 0 なら偽、0 以外なら真なので、if (strcmp(argv[i], opt)) が真になるのは文字列が違うときです。-q を探すと、argv[1] の "-q" とは等しいので条件は偽になり、ほかに要素はないので false を返して 0 と出ます。-x を探すと、"-q" と違うので条件が真になり、true を返して 1 と出ます。

Right

if (strcmp(argv[i], opt) == 0) {
    return true;
}

戻り値を 0 と比べれば、条件が真になるのは文字列が等しいときだけになります。-q は argv[1] と一致して true、-x はどの要素とも一致せずループを最後まで回って false を返すので、1 と 0 が出力されます。直す前のコードは、引数が2つ以上あると opt と違う要素が1つでもあれば true を返すので、オプションがあってもなくてもたいてい true になっていました。== 0 を書いておけば、引数の数によらず一致したときだけ true を返します1。

  1. strcmp は真偽ではなく大小を返すので、qsort に渡す比較関数の中でもそのまま使えます。C の規格は、等しくないときの値を負か正かだけ定めていて、-1 や 1 になるとは限りません。
