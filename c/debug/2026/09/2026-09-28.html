count_files.c

struct frame {
    dev_t dev;
    ino_t ino;
    const struct frame *up;
};

static int count_files(const char *path, const struct frame *up) {
    struct stat st;
    if (lstat(path, &st) != 0) return 0;
    if (S_ISREG(st.st_mode)) return 1;
    if (!S_ISDIR(st.st_mode)) return 0;
    for (const struct frame *f = up; f; f = f->up) {
        if (f->dev == st.st_dev && f->ino == st.st_ino) return 0;
    }
    struct frame me = {st.st_dev, st.st_ino, up};
    DIR *dir = opendir(path);
    if (!dir) return 0;
    int count = 0;
    struct dirent *e;
    while ((e = readdir(dir))) {
        if (strcmp(e->d_name, ".") == 0 || strcmp(e->d_name, "..") == 0) continue;
        char child[4096];
        snprintf(child, sizeof child, "%s/%s", path, e->d_name);
        count += count_files(child, &me);
    }
    closedir(dir);
    return count;
}

ディレクトリの下にある通常ファイルを、シンボリックリンクの先までたどって数える count_files です。祖先へ戻るリンクは循環になるので飛ばします。ディレクトリ data に通常ファイル a.txt、リンク shared -> ../shared、リンク self -> . があり、ディレクトリ shared には b.txt があります。count_files("data", NULL) はいま 1 を返しますが、本来は 2 です。どこを直せばよいでしょうか?

Answer

lstat を stat に変えます。

lstat は、パスがシンボリックリンクならリンク先ではなくリンクそのものの情報を返します。data/shared を調べると st_mode はシンボリックリンクを表すので、S_ISREG も S_ISDIR も偽になり、中に入らずに 0 が返ります。data/self も同じで、数えられるのは data/a.txt だけです。

stat はリンクをたどり、リンク先の情報を返します。data/shared はディレクトリ shared として扱われ、中の b.txt が数えられて 2 になります。data/self の先は data 自身ですが、data は祖先として up に積まれているので、そこで 0 が返ってループしません。

lstat のままではどのリンクの先にも入らないので、祖先の判定が働く場面はありませんでした。リンクをたどるようにした時点で、この判定が要るようになります。
