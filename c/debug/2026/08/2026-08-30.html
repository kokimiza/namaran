digest.c

#include <stdio.h>
#include <string.h>

#define DIGEST_LEN 4

static int digest_equal(const unsigned char* a, const unsigned char* b) {
    return strncmp((const char*)a, (const char*)b, DIGEST_LEN) == 0;
}

int main(void) {
    const unsigned char stored[DIGEST_LEN] = {0x5e, 0x00, 0x8c, 0x21};
    const unsigned char input[DIGEST_LEN] = {0x5e, 0x00, 0x13, 0xf7};
    puts(digest_equal(stored, input) ? "accepted" : "rejected");
}

パスワードをハッシュ関数にかけた4バイトの値(ダイジェスト)を、保存しておいた値と比べ、4バイトすべてが一致したときだけ受け付けるつもりのコードです。main は、保存した stored と、入力から計算した input を比べます。2つは3バイト目から違うので本来は rejected と表示されるはずが、いまは accepted と表示されます。

Answer

Wrong

return strncmp((const char*)a, (const char*)b, DIGEST_LEN) == 0;

strncmp は文字列を比べる関数で、最大 DIGEST_LEN 文字まで比べますが、その前に両方が同じ位置で 0 のバイト(文字列の終端)に来ると、そこで「一致」として 0 を返します。ダイジェストはハッシュ関数が出したバイト列なので、0 のバイトはどこにでも現れます。

stored と input は1バイト目の 0x5e が一致し、2バイト目はどちらも 0x00 です。strncmp はここで両方の文字列が終わったと見なし、違いのある3バイト目と4バイト目を読まずに 0 を返すので、digest_equal は 1 を返して accepted が出ます。先頭から 0 のバイトまでが保存値と同じダイジェストなら、残りが何であっても受け付けてしまいます。

Right

return memcmp(a, b, DIGEST_LEN) == 0;

memcmp は 0 のバイトを特別扱いせず、渡した長さの分だけバイトを比べます。DIGEST_LEN の4バイトすべてが比較に入るので、3バイト目の 0x8c と 0x13 の違いで 0 以外が返り、rejected が出ます。ダイジェストのように長さが決まったバイナリのデータは、終端の 0 を探す str の関数ではなく、長さを受け取る mem の関数で扱います。1

  1. 同じ欠陥の有名な例に、Wii の署名検証があります。署名から取り出したハッシュ値と、データから計算したハッシュ値を strncmp で比べていたため、先頭が 0 のバイトになるよう細工した偽の署名が検証を通りました。この穴は Trucha bug と呼ばれ、非公式のソフトを動かすのに使われました。
