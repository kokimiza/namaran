commit_row.rs

use std::cell::RefCell;

struct Row {
    version: u64,
    stock: i32,
}

fn commit(row: &RefCell<Row>, expected: u64, stock: i32) -> Result<u64, u64> {
    let current = row.borrow();
    if current.version != expected {
        return Err(current.version);
    }
    *row.borrow_mut() = Row { version: expected + 1, stock };
    Ok(expected + 1)
}

fn main() {
    let row = RefCell::new(Row { version: 3, stock: 10 });
    println!("{:?}", commit(&row, 2, 7));
    println!("{:?}", commit(&row, 3, 7));
}

commit は、行を読んだときの版番号 expected がいまの版番号と一致すれば在庫を書き換えて新しい版番号を Ok で返し、一致しなければ書き換えずにいまの版番号を Err で返すつもりの関数です。main は版番号 3 の行に対し、古い版 2 と正しい版 3 で1回ずつ呼びます。いまの出力は Err(3) の1行だけで、2回目の呼び出しが RefCell already borrowed の panic で止まります。本来の出力は Err(3) と Ok(4) の2行です。

Answer

Wrong

let current = row.borrow();
if current.version != expected {
    return Err(current.version);
}
*row.borrow_mut() = Row { version: expected + 1, stock };

row.borrow() が返す Ref は、共有の借用が続いていることを RefCell に知らせる番人の値で、捨てられるときに借用を返します。current という変数に入れた Ref が捨てられるのは、current を最後に使った行ではなく、関数のブロックの終わりです。参照の借用ならコンパイラが最後の使用までで終わらせますが、Drop を持つ値の破棄はスコープの終わりまで延びません。

版が一致しない1回目は、return で関数を抜けるときに current も捨てられるので、何も起きずに Err(3) が返ります。版が一致した2回目は、current を持ったまま borrow_mut() に進みます。共有の借用が残っている RefCell は可変の借用を許さないので、ここで panic します。必ず落ちるのは衝突しなかった更新のほうで、本来いちばん多く通るはずの道です。

Right

let current = row.borrow().version;
if current != expected {
    return Err(current);
}
*row.borrow_mut() = Row { version: expected + 1, stock };

版番号の u64 だけをコピーして取り出します。row.borrow() が返した Ref は文の中の一時的な値になり、let の文が終わったところで捨てられます。borrow_mut() に着いたときには借用が残っていないので、書き込みが通って Ok(4) が返ります。元のコードの if の後ろに drop(current); を置き、書き込む前に借用を返しても直ります。
