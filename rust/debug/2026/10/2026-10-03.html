inventory.rs

use std::cell::RefCell;

struct Inventory {
    stock: RefCell<Vec<u32>>,
}

impl Inventory {
    fn total(&self) -> u32 {
        self.stock.borrow().iter().sum()
    }
    fn restock(&self, n: u32) -> u32 {
        let mut stock = self.stock.borrow_mut();
        stock.push(n);
        self.total()
    }
}

fn main() {
    let inv = Inventory { stock: RefCell::new(vec![3]) };
    println!("{}", inv.restock(4));
}

Inventory::restock は、在庫に数量を1件追加し、追加したあとの合計を返すメソッドです。引数も標準入力も使わず、在庫 [3] に 4 を追加した結果を出力します。いまは何も出力されず、RefCell already mutably borrowed の panic で止まります。本来の出力は 7 です。

Answer

Wrong

let mut stock = self.stock.borrow_mut();
stock.push(n);
self.total()

borrow_mut() が返す RefMut はガードで、これが生きている間、RefCell は可変借用されたままになります。let で変数 stock に束縛したので、ガードは restock の末尾まで生きています。その途中で呼ぶ self.total() が self.stock.borrow() で読み取りの借用を求めますが、可変借用とは同時に成り立たないので panic します。

Right

self.stock.borrow_mut().push(n);
self.total()

ガードを変数に束縛せず、一時値のまま push を呼びます。一時値のガードはその文の終わりの ; で破棄されるので、self.total() を呼ぶ時点では可変借用が終わっています。変数 stock を残したまま、self.total() の前に drop(stock); と書いて明示的に手放しても直ります。
