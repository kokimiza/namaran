fn parse(csv: &str) -> Vec<Vec<String>> {
    csv.lines()
        .map(|line| {
            let (mut row, mut field, mut quoted) = (vec![], String::new(), false);
            for c in line.chars() {
                match c {
                    '"' => quoted = !quoted,
                    ',' if !quoted => row.push(std::mem::take(&mut field)),
                    _ => field.push(c),
                }
            }
            row.push(field);
            row
        })
        .collect()
}

lines は引用符を見ずに、すべての改行で文字列を切ります。"line1\nline2" の中の改行もここで切られ、1,"line1 と line2" が別々の行としてクロージャに渡ります。

quoted はクロージャの中で行ごとに false から作り直されるので、引用符の中にいるという状態も次の行へ引き継がれません。1,"line1 は開いた引用符を閉じないまま ["1", "line1"] になり、line2" は閉じるための引用符を開く引用符として読んで ["line2"] になります。

Right

fn parse(csv: &str) -> Vec<Vec<String>> {
    let (mut rows, mut row, mut field, mut quoted) = (vec![], vec![], String::new(), false);
    for c in csv.chars() {
        match c {
            '"' => quoted = !quoted,
            ',' if !quoted => row.push(std::mem::take(&mut field)),
            '\n' if !quoted => {
                row.push(std::mem::take(&mut field));
                rows.push(std::mem::take(&mut row));
            }
            _ => field.push(c),
        }
    }
    rows
}

文字列全体を1つのループで読み、quoted を行をまたいで持ち続けます。改行は引用符の外にあるときだけ行の区切りになり、中にある改行は _ の腕でフィールドに入ります。"line1\nline2" は1つのフィールドのまま残り、2行目は ["1", "line1\nline2"] になります1。

  1. 直した parse は各行が改行で終わる前提で、最後の行に改行がなければその行を返しません。ループのあとに残ったフィールドと行を足せば、改行で終わらない入力も読めます。
