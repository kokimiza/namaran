arena_cell.rs

use std::cell::RefCell;

struct Arena {
    items: RefCell<Vec<String>>,
}

impl Arena {
    fn alloc(&self, s: String) -> usize {
        let mut items = self.items.borrow_mut();
        items.push(s);
        items.len() - 1
    }
}

fn main() {
    let arena = Arena { items: RefCell::new(Vec::new()) };
    let a = arena.alloc(String::from("node"));
    let b = arena.alloc(arena.items.borrow()[a].clone() + "!");
    println!("{}", arena.items.borrow()[b]);
}

Arena は文字列をまとめて持ち、&self のまま新しい文字列を確保して添字を返すつもりのアリーナです。このプログラムは確保済みの "node" の末尾に ! を付けた文字列を確保して表示し、本来は node! と表示するはずです。いまは何も表示せず、alloc の borrow_mut の行で RefCell already borrowed と出て panic します。

Answer

Wrong

let b = arena.alloc(arena.items.borrow()[a].clone() + "!");

arena.items.borrow() は、借用が続いていることを示す Ref を返します。この Ref は名前を付けていない一時的な値で、let 文の中で作った一時的な値は、その文の終わりまで破棄されません。.clone() で文字列を複製しても、複製元を借りている Ref は残ったままです。

alloc の呼び出しは同じ文の中にあるので、alloc の中の borrow_mut は、Ref がまだ生きているうちに走ります。RefCell は共有の借用が残っている間の可変の借用を実行時に拒むので、ここで panic します。コンパイラは RefCell の中の借用を追わないので、コンパイルは通ります。

Right

let name = arena.items.borrow()[a].clone() + "!";
let b = arena.alloc(name);

読む文と確保する文を分けます。1つ目の文の終わりで Ref が破棄されて借用が終わるので、2つ目の文の borrow_mut は成功します。name は複製した String を所有していて、Ref を借りていないので、借用が終わっても使えます。1

  1. clone() を外して let name = &arena.items.borrow()[a]; と書くと、今度は Ref の寿命が name と同じだけ延びるので、次の文の alloc でやはり panic します。
