withdraw.rs

use std::sync::Mutex;
use std::thread;
use std::time::Duration;

fn withdraw(balance: &Mutex<i64>, amount: i64) -> bool {
    if *balance.lock().unwrap() < amount {
        return false;
    }
    thread::sleep(Duration::from_millis(100)); // ask the fraud checker
    *balance.lock().unwrap() -= amount;
    true
}

fn main() {
    let balance = Mutex::new(100);
    thread::scope(|s| {
        s.spawn(|| withdraw(&balance, 70));
        s.spawn(|| withdraw(&balance, 70));
    });
    println!("{}", balance.into_inner().unwrap());
}

withdraw は、残高が足りるときだけ amount を引いて true を返すつもりの関数です。入力は main のとおりで、残高 100 から2つのスレッドが同時に 70 ずつ引き出そうとし、最後の残高を表示します。いまは -40 が出ます(スレッドの進み方しだいで 30 になることもあります)。本来の出力は、片方だけが成功した 30 です。どこを直せばよいでしょうか?

Answer

if の条件で作った MutexGuard は一時値なので、条件を評価し終えた時点で破棄され、ロックが外れます。そのあと sleep している間は誰もロックを持っておらず、もう1つのスレッドも残高 100 を見て確認を通ります。両方が確認を終えてから両方が 70 を引くので、残高は -40 です。確認した時点の値が使う時点でも正しいと思い込むこの種の競合を、TOCTOU(time of check to time of use)と呼びます。

直すには、ロックを1回だけ取り、そのガードで確認から更新までを守ります: 

    let mut balance = balance.lock().unwrap();
    if *balance < amount {
        return false;
    }
    thread::sleep(Duration::from_millis(100)); // ask the fraud checker
    *balance -= amount;

変数に束縛したガードは、関数を抜けるまで破棄されません。後から来たスレッドは lock() で待たされ、先のスレッドが 30 に減らしたあとの残高を見て false を返します。元のコードも1回ずつのロックは正しく取れていて、データ競合は起きていないので、コンパイラはこの誤りを見つけられません。
