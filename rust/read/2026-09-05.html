ttl_cache.rs

use std::collections::HashMap;

const TTL: u64 = 10;

fn get(cache: &HashMap<&str, (i32, u64)>, key: &str, now: u64) -> Option<i32> {
    cache.get(key).filter(|&&(_, exp)| now < exp).map(|&(v, _)| v)
}

fn main() {
    let mut cache = HashMap::new();
    for (now, key, value) in [(0, "a", 1), (5, "b", 2), (8, "a", 3)] {
        cache.entry(key).or_insert((value, now + TTL));
    }
    println!("{:?} {:?}", get(&cache, "a", 12), get(&cache, "b", 12));
}

これは何を出力するでしょうか?

Answer

None Some(2) です。

entry(key).or_insert(...) は、キーがまだないときだけ値を入れます。キーがすでにあれば何も書き換えず、今ある値への可変参照を返すだけです。時刻 8 の "a" を入れる時点では、時刻 0 に入れた "a" が残っているので、値 3 も期限 18 も入らず、"a" は (1, 10) のままです。

get は now < exp のときだけ値を返します。時刻 12 では "a" の期限 10 を過ぎているので None、"b" は期限 15 より前なので Some(2) です。入れ直すたびに期限を延ばしたいなら、cache.insert(key, (value, now + TTL)) で上書きします。
