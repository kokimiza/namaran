merge_into.rs

use std::cell::RefCell;
use std::rc::Rc;

fn merge_into(dst: &Rc<RefCell<Vec<i32>>>, src: &Rc<RefCell<Vec<i32>>>) {
}

src の要素をすべて順に dst の末尾へ移し、src を空にする merge_into を書いてください。dst と src に同じリストを指す Rc が渡されることもあり、そのときは何も変えずに戻ります。src が空なら dst はそのままです。出力例は、呼び出したあとの a と b の中身です。

入力例 1

let a = Rc::new(RefCell::new(vec![1, 2])); let b = Rc::new(RefCell::new(vec![3])); merge_into(&a, &b);

出力例 1

[1, 2, 3] []

入力例 2

let a = Rc::new(RefCell::new(vec![1, 2])); let b = Rc::clone(&a); merge_into(&a, &b);

出力例 2

[1, 2] [1, 2]

入力例 3

let a = Rc::new(RefCell::new(vec![1])); let b = Rc::new(RefCell::new(Vec::new())); merge_into(&a, &b);

出力例 3

[1] []
Reference 
use std::cell::RefCell;
use std::rc::Rc;

fn merge_into(dst: &Rc<RefCell<Vec<i32>>>, src: &Rc<RefCell<Vec<i32>>>) {
    if Rc::ptr_eq(dst, src) {
        return;
    }
    dst.borrow_mut().append(&mut src.borrow_mut());
}

最後の1行だけでも入力例1と3は動きますが、入力例2は RefCell already borrowed の panic で止まります。Rc::clone で作った a と b は同じ RefCell を指しており、どちらも &Rc<…> という共有参照なので、同じものを2つ渡しても借用検査は何も言いません。RefCell はその代わりに、借用を実行時に数えます。

borrow_mut() が返す RefMut は一時値で、一時値はその文の終わりまで破棄されません。dst.borrow_mut() を持ったまま同じ文の中で src.borrow_mut() を呼ぶと、同じ RefCell に可変借用が2つ同時に立とうとし、2つ目が panic します。let で2つの RefMut を順に変数へ取っても、同じ理由で落ちます。

参考実装は、先に Rc::ptr_eq で2つの Rc が同じ割り当てを指しているかを確かめ、同じなら何もせずに戻ります。同じリストの要素を自分の末尾へ移しても中身は変わらないので、これが求める結果です。dst == src と書くと、Rc の == は指している中身の Vec どうしを比べるので、内容がたまたま等しい別のリストまで同じものとみなしてしまいます。別のリストなら2つの RefCell は別物で、同時に可変借用しても衝突しません。Vec::append は src の要素を移し終えると src を空にします。

掲載したのは参考実装で、唯一の正解ではありません。let moved = std::mem::take(&mut *src.borrow_mut()); で先に src の中身を取り出し、その文で借用を終えてから dst.borrow_mut().extend(moved); とする書き方なら、同じリストが渡されても panic せず、取り出した要素がそのまま戻ります。
