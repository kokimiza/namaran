lock_file.rs

use std::io;
use std::path::Path;

fn try_acquire(path: &Path) -> io::Result<bool> {
}

path にロックファイルを作る try_acquire を書いてください。ファイルを新しく作れたら Ok(true)、すでにあれば Ok(false) を返し、それ以外の I/O エラーは Err で返します。複数のプロセスが同時に呼んでも、Ok(true) を受け取るのは1つだけでなければなりません。空のディレクトリで try_acquire(Path::new("app.lock")) → Ok(true)、続けて同じ呼び出し → Ok(false)、try_acquire(Path::new("no/such/dir/app.lock")) → 種類が NotFound の Err

use std::fs::OpenOptions;
use std::io;
use std::path::Path;

fn try_acquire(path: &Path) -> io::Result<bool> {
    match OpenOptions::new().write(true).create_new(true).open(path) {
        Ok(_) => Ok(true),
        Err(e) if e.kind() == io::ErrorKind::AlreadyExists => Ok(false),
        Err(e) => Err(e),
    }
}

path.exists() で確かめてから File::create で作ると、確かめてから作るまでの間に、別のプロセスが同じファイルを作れます。両方が「無い」と確かめたあとで両方が作れば、どちらも Ok(true) を受け取ります。File::create は既存のファイルを空にして開くだけで失敗しないので、この競合はエラーとしても表に出ません。

OpenOptions の create_new(true) は、ファイルが無ければ作り、あれば AlreadyExists のエラーを返します。存在の確認と作成を OS が1回の操作として行うので、間に割り込まれることがありません1。create_new でファイルを作るには、write(true) か append(true) も指定しておく必要があります。あとはエラーの種類で分け、AlreadyExists だけを Ok(false) に変えて、ディレクトリが無いときの NotFound や権限のエラーはそのまま返します。

掲載したのは参考実装で、唯一の正解ではありません。match の代わりに map と or_else をつないで書くこともできます。

  1. Unix では open に O_CREAT | O_EXCL を付けた呼び出し、Windows では CreateFileW に CREATE_NEW を渡した呼び出しに当たります。
