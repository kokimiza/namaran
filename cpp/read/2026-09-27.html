spawn_echo.cpp

#include <spawn.h>
#include <string>
#include <sys/wait.h>
#include <vector>

extern char **environ;

int run(const std::string &file, std::vector<std::string> args) {
    std::vector<char *> argv;
    for (auto &a : args) {
        argv.push_back(a.data());
    }
    argv.push_back(nullptr);

    pid_t pid;
    if (posix_spawnp(&pid, file.c_str(), nullptr, nullptr, argv.data(), environ) != 0) {
        return -1;
    }
    int status = 0;
    waitpid(pid, &status, 0);
    return status;
}

int main() {
    run("echo", {"hello", "world", "*"});
    run("echo", {"echo", "$HOME", "~"});
}

これは何を出力するでしょうか?

Answer

1行目が world *、2行目が $HOME ~ です。

1行目から hello が消えるのは、args の先頭がそのまま子プロセスの argv[0] になるからです。argv[0] はプログラム自身の名前を入れる場所で、echo はここを表示しません。posix_spawnp の第2引数 file は実行するファイルを探すためだけに使われ、argv[0] を埋めてはくれません。シェルで echo hello world と打てばシェルが {"echo", "hello", "world"} を組み立てますが、自分で argv を作るときは先頭に自分でプログラム名を置きます。2回目の呼び出しが "echo" から始まっているのはそのためです。

残りの引数には、書いた文字列がそのまま届きます。* をファイル名の一覧に、$HOME を環境変数の値に、~ をホームディレクトリに置き換えるのはシェルの仕事です。posix_spawnp はシェルを通さずに echo を直接起動し、echo 自身はこれらの記号を特別扱いしないので、受け取った文字列を空白でつないで表示します。

同じ理由で、引数に空白や ; が入っていても1つの引数のまま届き、別のコマンドとして実行されることはありません。外から受け取ったファイル名などを渡すときに、シェルを通さない起動が選ばれるのはこのためです。
