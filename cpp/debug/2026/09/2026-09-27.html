bracket_names.cpp

#include <spawn.h>
#include <string>
#include <sys/wait.h>
#include <vector>

extern char **environ;

int run(std::vector<std::string> args) {
    std::vector<char *> argv;
    for (auto &a : args) {
        argv.push_back(a.data());
    }
    argv.push_back(nullptr);

    pid_t pid;
    if (posix_spawnp(&pid, argv[0], nullptr, nullptr, argv.data(), environ) != 0) {
        return -1;
    }
    int status = 0;
    waitpid(pid, &status, 0);
    return status;
}

int main() {
    std::string name = "my notes.txt";
    run({"printf", "[%s]\n", "\"" + name + "\"", "todo.txt"});
}

run は、受け取った引数の並びでプログラムをシェルを通さずに起動する関数です。これを使い、printf で2つのファイル名を1つずつ [ と ] で囲んで表示します。入力のファイル名は name の my notes.txt と todo.txt です。いまの出力は ["my notes.txt"] と [todo.txt] の2行ですが、本来の出力は [my notes.txt] と [todo.txt] です。どこを直せばよいでしょうか?

Answer

name の前後に足している引用符を外し、name をそのまま渡します: 

    run({"printf", "[%s]\n", name, "todo.txt"});

シェルでは、空白を含むファイル名を1つの引数にするために "my notes.txt" と引用符で囲みます。この引用符は、シェルが引数の区切りを決めるための記号です。区切りが決まったあとでシェルが取り除くので、プログラムには my notes.txt だけが届きます。

run はシェルを通さないので、引数の区切りは std::vector の要素の区切りそのものです。name は最初から1つの要素なので、空白が入っていても2つには分かれません。そこへ引用符を足すと、取り除く役目のシェルがいないので、引用符も文字列の一部として printf に届きます。
