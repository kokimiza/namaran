#include <iostream>
#include <map>
#include <string>

std::map<std::string, std::string> conf;

void set(std::string key,
         std::string value) {
    conf[key] = value;
}
void set(std::string key, bool on = true) {
    conf[key] = on ? "yes" : "no";
}

int main() {
    set("name", "bob");
    std::cout << conf["name"] << '\n';
}

set は、設定の表 conf にキーと値の文字列を記録するつもりの関数です。値を省いた set("verbose") でフラグを立てられるように、bool を受け取り、"yes" か "no" を記録するオーバーロードも用意してあります。main はキー name に bob を記録し、その値を出力します。main の呼び出しは変えずに直してください。

void set(std::string key,
         std::string value) {
    conf[key] = value;
}
void set(std::string key, bool on = true) {
    conf[key] = on ? "yes" : "no";
}

set("name", "bob") の "bob" は const char[4] の配列で、std::string ではありません。1つ目の set へ渡すには、std::string のコンストラクタを通すユーザー定義変換が要ります。2つ目の set へは、配列からポインタへ、ポインタから bool へという標準変換だけで渡せます。オーバーロード解決は標準変換だけで済む候補を優先するので、呼ばれるのは bool の set です。NULL でないポインタは true になり、name には "yes" が記録されます。

Right 時間計算量 𝒪(m log n) 空間計算量 𝒪(m) n は conf のキーの個数、m はキーと値の文字数の合計 

void set(std::string key,
         std::string value) {
    conf[key] = value;
}
void set(std::string key,
         const char* value) {
    set(key, std::string(value));
}
void set(std::string key, bool on = true) {
    conf[key] = on ? "yes" : "no";
}

const char* を受け取る set を足すと、"bob" は配列からポインタへの変換だけでこの候補に渡せます。bool の候補はそのうえにポインタから bool への変換が要るので、新しい候補のほうが優先されます1。足した set は std::string を作って1つ目の set に渡すだけです。値を省いた set("verbose") は、引数2つの候補では呼べないので、これまでどおり bool の set に既定の値 true で解決されます。

  1. 呼び出し側を変えてよいなら、std::string_literals の "bob"s と書いて最初から std::string を渡す方法もあります。std::string_view を受け取る候補を足しても、std::string_view への変換はユーザー定義変換なので、bool の候補には勝てません。
