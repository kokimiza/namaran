square_size.zig

const std = @import("std");

const Size = struct {
    w: u8 = 4,
    h: u8 = 3,
    const square: Size = .{ .h = 4 };
};

pub fn main(init: std.process.Init) !void {
    const s: Size = .square;
    var buf: [64]u8 = undefined;
    const stdout = std.Io.File.stdout();
    var w = stdout.writer(init.io, &buf);
    const out = &w.interface;
    try out.print("{d}\n", .{s.w * s.h});
    try out.flush();
}

これは何を出力するでしょうか?

Answer

Output

16

const s: Size = .square; の右辺は、型が Size と決まっている場所に書いた .square なので、Size の中の宣言 Size.square を指します。langref はこの書き方を decl literal と呼びます。構造体は、フィールドのほかに定数や関数を宣言できる名前空間でもあり、square はインスタンスごとのフィールドではなく、型に1つだけある定数です。

square の値 .{ .h = 4 } は h だけを書いた初期化なので、書かなかった w にはフィールドの既定値 4 が入ります。s は w = 4、h = 4 になり、s.w * s.h は 16 です。

.square を、まだ何も書いていない .{} と同じように読むと、既定値どうしの 4 × 3 = 12 と答えることになります。.{ .h = 4 } の w を 0 と読むと 0 です。
