sort_digit.zig

const std = @import("std");

fn byLastDigit(_: void, a: u8, b: u8) bool {
    return a % 10 < b % 10;
}

pub fn main(init: std.process.Init) !void {
    var xs = [_]u8{ 23, 41, 15, 32 };
    std.mem.sort(u8, &xs, {}, byLastDigit);
    var buf: [64]u8 = undefined;
    const stdout = std.Io.File.stdout();
    var w = stdout.writer(init.io, &buf);
    const out = &w.interface;
    for (xs) |x| try out.print("{d} ", .{x});
    try out.print("\n", .{});
    try out.flush();
}

これは何を出力するでしょうか?

Answer

Output

41 32 23 15

std.mem.sort は、要素の型、並べ替えるスライス、比較関数に渡す文脈の値、比較関数の4つを受け取ります。比較関数は、文脈と2つの要素 a、b を受け取り、a を b より前に置くべきときに true を返す、という約束で呼ばれます。並べ替えはこの答えだけを頼りに要素を動かすので、何の大小で並ぶかは比較関数の中身で決まります。ここでは文脈は使わないので、空の値 {}（型は void）を渡し、比較関数の側では _ で受けて捨てています。

byLastDigit は a % 10 と b % 10 を比べるので、前後は一の位だけで決まります。23、41、15、32 の一の位は 3、1、5、2 なので、一の位の小さい順に 41、32、23、15 と並びます。数そのものの大きさでは並ばないので、15、23、32、41 にはなりません。一の位が同じ数は含まれないので、比較で前後が決まらない要素の扱いは結果に関係しません。
