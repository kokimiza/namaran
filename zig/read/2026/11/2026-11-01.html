limits.zig

const std = @import("std");

const Range = struct {
    min: u8 = 0,
    max: u8 = 100,
    const init: Range = .{ .min = 1, .max = 10 };
};

const Config = struct {
    range: Range = .init,
};

pub fn main(init: std.process.Init) !void {
    const a: Config = .{};
    const b: Config = .{ .range = .{ .max = 50 } };
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    try stdout.print("{d} {d} {d} {d}\n", .{ a.range.min, a.range.max, b.range.min, b.range.max });
    try stdout.flush();
}

これは何を出力するでしょうか?

Answer

1 10 0 50 です。

a は .{} で、range を指定していないので、range にはフィールドのデフォルト値 .init が入ります。.init は、結果の型 Range の中にある宣言 Range.init を指す書き方です。Range.init は .{ .min = 1, .max = 10 } なので、a は 1 10 です。.init を「デフォルト値で初期化する」意味に読むと、0 100 と答えます。

b は range を .{ .max = 50 } と指定しています。指定したフィールドにはデフォルト値の .init は使われず、この構造体リテラルが range の値をまるごと決めます。リテラルで省いた min に入るのは、Range のフィールド min のデフォルト値 0 です。Range.init の min の 1 は引き継がれません。デフォルト値の .init に .max だけを上書きすると読むと、b を 1 50 と答えます。
