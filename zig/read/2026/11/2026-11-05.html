labels.zig

const std = @import("std");

const Item = struct {
    id: u32,
    var buf: [16]u8 = undefined;
    fn label(self: Item) ![]const u8 {
        return std.mem.print(&buf, "item-{d}", .{self.id});
    }
};

pub fn main(init: std.process.Init) !void {
    var out: [64]u8 = undefined;
    var w = std.Io.File.stdout().writer(init.io, &out);
    const a = try Item.label(.{ .id = 7 });
    const b = try Item.label(.{ .id = 12 });
    try w.interface.print("{s} {s}\n", .{ a, b });
    try w.interface.flush();
}

これは何を出力するでしょうか?

Answer

Output

item-1 item-12

Item の中の id: u32, はフィールドで、値ごとに1つずつあります。var buf: [16]u8 = undefined; はフィールドではなく、Item という名前空間に置いた変数（container-level variable）です。値をいくつ作っても buf はプログラム全体で1つしかなく、label は呼ばれるたびに同じ buf の先頭から書きます。Item 型の値 x があれば、Item.label(x) と x.label() は同じ呼び出しです。メソッドは、名前空間にある関数を値から呼べるようにしたものにすぎません。

std.mem.print は文字列をコピーして返すのではなく、書いた範囲を指す buf のスライスを返します。スライスは先頭の位置と長さの組なので、a は buf の先頭から6バイトを指します。この時点の中身は item-7 です。続く label が同じ buf の先頭に item-12 を書くと、b は先頭から7バイトを指します。a の長さは6のままなので、上書きされた buf の先頭6バイト item-1 を見ることになります。

buf を値ごとにあるものと読むと item-7 item-12 になり、2つのスライスが同じ場所を指すことに気付いても、長さまで新しい文字列に合わせて変わると考えると item-12 item-12 になります。
