skip_rows.zig

const std = @import("std");
const rows = [_][]const u8{ "a #", "#", "" };

pub fn main(init: std.process.Init) !void {
    var buf: [64]u8 = undefined;
    const stdout = std.Io.File.stdout();
    var w = stdout.writer(init.io, &buf);
    const out = &w.interface;
    outer: for (rows) |row| {
        for (row) |c| {
            if (c == '#') continue :outer;
            if (c == ' ') break;
        }
        try out.print("[{s}]", .{row});
    }
    try out.print("\n", .{});
    try out.flush();
}

これは何を出力するでしょうか?

Answer

Output

[a #][]

ラベルの無い break と continue は、それを囲む一番内側のループに効きます。外側のループに効かせるには、ループに outer: のようなラベルを付け、continue :outer のようにラベルを指定します。このコードでは、continue :outer は外側の for の次の行へ進み、break は内側の for (row) だけを抜けます。

1行目の "a #" では、'a' はどちらの条件にも当たらず、次の空白で break します。抜けるのは内側のループだけなので、外側のループの本体はその下の print へ進み、[a #] が出ます。空白の後ろの '#' は調べられません。2行目の "#" では最初の文字で continue :outer が実行され、外側の本体の残りにある print を飛ばして3行目へ進みます。3行目の "" は長さが 0 なので、内側のループは1回も回らずに終わり、そのまま print で [] が出ます。

break が外側のループまで抜けると読むと、何も出力されないと答えてしまいます。continue :outer を内側のループの continue と同じに読むと、2行目も出力されて [a #][#][] になります。どちらの規則も分かっていても、空の行では内側のループが何もせずに終わることを見落とすと、[a #] と答えることになります。
