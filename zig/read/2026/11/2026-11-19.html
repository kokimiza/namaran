same_width.zig

const std = @import("std");
const count = std.fmt.count;

pub fn main(init: std.process.Init) !void {
    var buf: [64]u8 = undefined;
    const stdout = std.Io.File.stdout();
    var w = stdout.writer(init.io, &buf);
    const out = &w.interface;
    var n: u32 = 0;
    for (0..1000) |i| {
        const hex = count("{x}", .{i});
        const dec = count("{d}", .{i});
        if (hex == dec) n += 1;
    }
    try out.print("{d}\n", .{n});
    try out.flush();
}

これは何を出力するでしょうか?

Answer

Output

838

std.fmt.count は、書式と引数から作られる文字列のバイト数を、どこにも書き出さずに返します。{x} は小文字の16進数を 0x も先頭の 0 も付けずに書き、{d} は10進数を書くので、hex == dec は「16進で書いた桁数と10進で書いた桁数が等しい」という条件です。0 はどちらでも 0 の1文字です。

1000回を1つずつ追う必要はありません。桁数が変わるのは、10進では 10 と 100、16進では 16 と 256 のところだけで、そのあいだでは両方の桁数が一定です。この4つの境目で 0〜999 を区切ると、0〜9 は両方1桁で 10 個、10〜15 は16進では a〜f の1桁なのに10進は2桁、16〜99 は両方2桁で 84 個、100〜255 は16進では ff までの2桁なのに10進は3桁、256〜999 は両方3桁で 744 個です。等しい区間を足すと 10 + 84 + 744 = 838 になります。16進が4桁になるのは 4096 からで、この範囲には入りません。

16 は 10 より大きいので、16進の桁数が10進より多くなることはありません。食い違うのは、10進が先に桁を増やし、16進がまだ追い付いていない 10〜15 と 100〜255 だけです。この2つの区間を見落とすと 1000 と答え、1桁どうしの 0〜9 しか一致しないと考えると 10 と答えることになります。
