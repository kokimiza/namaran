opcodes.zig

const std = @import("std");

const Op = enum(u8) { inc = 2, dbl, halt = 0 };

pub fn main(init: std.process.Init) !void {
    var out: [64]u8 = undefined;
    var w = std.Io.File.stdout().writer(init.io, &out);
    var acc: u32 = 1;
    for ([_]u8{ 3, 2, 0, 3, 2 }) |byte| {
        switch (@as(Op, @fromBackingInt(byte))) {
            .inc => acc += 1,
            .dbl => acc *= 2,
            .halt => break,
        }
    }
    try w.interface.print("{d}\n", .{acc});
    try w.interface.flush();
}

これは何を出力するでしょうか?

Answer

Output

3

値を書かなかった enum のフィールドは、直前のフィールドの値に1を足した値になります。inc は2なので、その次の dbl は3です。halt は0と書いてあります。@fromBackingInt は整数を、その値を持つフィールドに読み替える組み込み関数なので、バイト列 3, 2, 0, 3, 2 は dbl、inc、halt、dbl、inc の順の命令になります。

Zig の switch は、腕を実行し終えると次の腕へ進まずに抜けるので、C のように腕の終わりを break で区切る必要がありません。break の対象になるのはループとラベルを付けたブロックで、ラベルの無い switch はなりません。そのため .halt => break は for 文を抜けます。acc は1から始まり、dbl で2、inc で3になったところで halt がループを終わらせ、残りの2バイトは読まれずに 3 が表示されます。

C の習慣で break が switch だけを抜けると読むと、halt のあとも続けて dbl で6、inc で7になり、7 を答えます。値を書かなかった dbl を、宣言の位置から1番目の1と読むと、3に当たるフィールドが無いので、最初のバイトで @fromBackingInt が実行時の安全検査（langref の言葉では safety-checked illegal behavior）に引っかかって止まる、と答えることになります。
