visit.zig

const std = @import("std");

const Visit = struct {
    var count: u32 = 0;
    fn log(_: Visit) void {
        count += 1;
    }
};

pub fn main(init: std.process.Init) !void {
    for ([_]Visit{ .{}, .{} }) |v| v.log();
    var buf: [64]u8 = undefined;
    const stdout = std.Io.File.stdout();
    var w = stdout.writer(init.io, &buf);
    const out = &w.interface;
    try out.print("{d}\n", .{Visit.count});
    try out.flush();
}

これは何を出力するでしょうか?

Answer

Output

2

var count: u32 = 0; は Visit の波括弧の中に書いてありますが、フィールドではありません。フィールドは count: u32 = 0, のように名前から書き始めてカンマで終え、値を作るたびにその値の中に1つずつ用意されます。var や const で始まる行は、Visit という名前空間に置いた宣言で、var ならプログラム全体で1つだけの変数になります（langref では container level variable と呼びます）。Visit にはフィールドが1つも無いので、Visit の値はどれも中身を持たず、count はどの値にも属しません。

for の |v| は配列の要素をコピーした値で、v.log() は Visit.log(v) と同じ呼び出しです。log は受け取った値を _ で捨て、名前空間の count に 1 を足します。要素は2つなので log は2回呼ばれ、どちらも同じ1つの count を増やすので、Visit.count は 2 です。

for の値キャプチャはコピーなので、メソッドの中で変えたものは配列に残らないと考えると、0 と答えることになります。コピーで変更が失われるのは、変えたものがその値のフィールドだった場合です。ここで増やしているのはコピーの中身ではなく名前空間の変数なので、どの値から呼んでも、値をコピーしても、増える先は同じです。
