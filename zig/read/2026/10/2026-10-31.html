counter.zig

const std = @import("std");

const Counter = struct {
    var total: u32 = 0;
    n: u32 = 0,
    fn hit(self: *Counter) void {
        self.n += 1;
        total += 1;
    }
};

pub fn main(init: std.process.Init) !void {
    var c: Counter = .{};
    c.hit();
    c = .{};
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    try stdout.print("{d} {d}\n", .{ c.n, Counter.total });
    try stdout.flush();
}

これは何を出力するでしょうか?

Answer

0 1 です。

構造体の中の n: u32 = 0, はフィールドで、Counter の値はそれぞれ自分の n を持ちます。var total: u32 = 0; のほうは宣言で、langref ではコンテナレベル変数(container level variable)と呼ばれます。total は Counter という名前空間に1つだけある変数で、どの値の一部でもなく、= 0 の初期化もプログラムの開始時に1回だけ行われます。

c.hit() は Counter.hit(&c) と同じ呼び出しで、c.n と total がどちらも1になります。次の c = .{}; は、フィールドの既定値から作った新しい値を c に代入します。c.n は0に戻りますが、total は c の値に含まれないので、1のまま変わりません。

var total: u32 = 0; を、直後の n: u32 = 0, と同じ「既定値つきのメンバー」と読むと、c = .{} で total も0に戻ると考えて 0 0 と答えてしまいます。
