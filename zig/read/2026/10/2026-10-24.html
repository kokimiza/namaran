headroom.zig

const std = @import("std");

const Reading = union(enum) {
    small: u8,
    large: u16,
};

fn headroom(r: Reading) u32 {
    return switch (r) {
        inline else => |v| std.math.maxInt(@TypeOf(v)) - v,
    };
}

pub fn main(init: std.process.Init) !void {
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    try stdout.print("{d} {d}\n", .{ headroom(.{ .small = 200 }), headroom(.{ .large = 200 }) });
    try stdout.flush();
}

これは何を出力するでしょうか?

Answer

55 65335 です。

inline else は、残りのタグ1つごとに腕(prong)をコンパイル時に展開します。この switch は、.small => |v| ... と .large => |v| ... の2本の腕を書いたのと同じで、それぞれの腕の v はそのタグの型を持ちます。.small の腕では v が u8 なので、std.math.maxInt(@TypeOf(v)) は 255 で、255 - 200 は 55 です。.large の腕では v が u16 なので上限は 65535 になり、65535 - 200 で 65335 です。

渡した値はどちらも 200 ですが、上限は腕ごとの型で決まります。関数の返り値の型 u32 は、腕の式を計算した結果を広げるときに使われるだけで、@TypeOf(v) が u32 になることはありません。返り値の型から上限を読むと、2つとも 4294967095 と答えることになります。
