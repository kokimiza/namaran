type_of.zig

const std = @import("std");

var calls: u32 = 0;

fn next() u16 {
    calls += 1;
    return 1;
}

pub fn main(init: std.process.Init) !void {
    const T = @TypeOf(next(), @as(u8, 0));
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    try stdout.print("{d} {d}\n", .{ @typeInfo(T).int.bits, calls });
    try stdout.flush();
}

これは何を出力するでしょうか?

Answer

16 0 です。

@TypeOf は式をいくつでも受け取り、それらを1つの型にそろえたときの型を返します。langref はこのそろえ方を Peer Type Resolution と呼んでいます。next() の型は u16、@as(u8, 0) の型は u8 で、u8 の値はどれも u16 で表せるので、T は u16 になります。@typeInfo(T) は型の情報をコンパイル時に返し、整数型なら .int にビット数 bits を持っています。1つ目の値は16です。

2つ目の calls は0のままです。langref は @TypeOf の引数について、式は評価するが実行時の副作用(runtime side-effects)が起きないことを保証しています。next() は型を決めるために調べられるだけで、プログラムの実行中に呼ばれることはなく、calls += 1 は一度も実行されません。引数の中の関数呼び出しが実際に呼ばれると読むと、16 1 と答えることになります。
