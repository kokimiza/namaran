rolled_back.zig

const std = @import("std");

var rolled_back: u32 = 0;

fn parsePort(s: []const u8) !u16 {
    errdefer rolled_back += 1;
    const n = try std.fmt.parseInt(u16, s, 10);
    if (n == 0) return error.Zero;
    return n;
}

pub fn main(init: std.process.Init) !void {
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    for ([_][]const u8{ "8080", "0", "x" }) |s| {
        if (parsePort(s)) |p| try stdout.print("{d} ", .{p}) else |err| try stdout.print("{s} ", .{@errorName(err)});
    }
    try stdout.print("{d}\n", .{rolled_back});
    try stdout.flush();
}

これは何を出力するでしょうか?

Answer

8080 Zero InvalidCharacter 2 です。

errdefer は、関数がエラーを返して終わるときにだけ、置いた式を実行します。エラーがどこから来たかは問わないので、try が parseInt のエラーをそのまま返す場合も、return error.Zero で返す場合も、rolled_back += 1 が実行されます。

"8080" は数として読めて 0 でもないので、parsePort は成功し、errdefer は実行されません。"0" は読めますが error.Zero を返し、"x" は parseInt が error.InvalidCharacter を返して try で抜けます。この2回だけ rolled_back が増えるので、最後に 2 が出ます。

Zig 0.17 の errdefer は、返そうとしているエラーの値を受け取れません。どのエラーだったかを使いたいときは、この main のように、呼び出し側で if の else |err| か catch |err| を使って受け取ります。1

  1. 以前の版の Zig には、errdefer |err| と書いて返すエラーを受け取る構文がありました。Zig 0.17 の文法では errdefer の後ろに置けるのは式だけで、この構文はコンパイルエラーになります。
