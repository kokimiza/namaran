skip_threes.zig

const std = @import("std");

pub fn main(init: std.process.Init) !void {
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    var i: u32 = 0;
    while (i < 10) : (i += 1) {
        if (i % 3 == 0) {
            i += 1;
            continue;
        }
        try stdout.print("{d} ", .{i});
    }
    try stdout.print("{d}\n", .{i});
    try stdout.flush();
}

これは何を出力するでしょうか?

Answer

2 5 8 11 です。

while (i < 10) : (i += 1) の : (i += 1) は継続式(continue expression)で、本体を最後まで実行したときだけでなく、continue で次の回へ進むときにも実行されます。i が3の倍数の回は、本体で i += 1 をしてから continue するので、継続式と合わせて i が2つ進みます。

i が0の回は1を足して continue し、継続式で2になります。2は表示され、継続式で3になります。3の回は同じように5へ飛ぶので、表示されるのは2、5、8です。8の次の9も3の倍数で、本体で10になったあとに継続式でさらに1を足すので、条件 i < 10 を確かめる時点の i は11です。ループを抜けたあとの行は、この11を表示します。

continue では継続式が飛ばされると読むと、本体の i += 1 だけで次の回へ進むので、1 2 4 5 7 8 10 と答えることになります。条件が偽になるのは i が10のときだと決めてかかると、最後の数を10と読み違えます。
