const std = @import("std");

pub fn main(init: std.process.Init) !void {
    var out: [64]u8 = undefined;
    var w = std.Io.File.stdout().writer(init.io, &out);
    const csv = ",a,,b,";
    var split = std.mem.splitScalar(u8, csv, ',');
    var tokens = std.mem.tokenizeScalar(u8, csv, ',');
    var n: usize = 0;
    var m: usize = 0;
    while (split.next()) |_| n += 1;
    while (tokens.next()) |_| m += 1;
    try w.interface.print("{d} {d}\n", .{ n, m });
    try w.interface.flush();
}

これは何を出力するでしょうか?

Answer

Output

5 2

splitScalar は区切り文字の , を欄の境目として扱い、境目と境目の間を、空であってもすべて1つの欄として返します。,a,,b, には , が4個あるので、欄は1つ多い5個です。先頭の , の前、a と b の間の2個の , に挟まれた所、末尾の , の後ろがそれぞれ空の欄になり、""、"a"、""、"b"、"" の順に返ります。最後の "" を返したあとで next が null になり、while を抜けます。

tokenizeScalar は区切り文字の並びをまとめて読み飛ばし、区切り文字でない文字の並びだけを返すので、"a" と "b" の2個です。splitScalar も空の欄を飛ばすと考えると 2 2、末尾の , の後ろだけは欄にならないと考えると 4 2 と答えます。CSV のように空の欄にも意味がある形式では splitScalar を、空白が何個続いても1つの区切りとみなしたい形式では tokenizeScalar を使います。
