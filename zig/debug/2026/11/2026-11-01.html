line.zig

const std = @import("std");

const Line = struct {
    bytes: [8]u8 = undefined,
    len: usize = 0,

    fn append(self: *Line, byte: u8) void {
        self.bytes[self.len] = byte;
        self.len += 1;
    }
};

pub fn main(init: std.process.Init) !void {
    var line: Line = undefined;
    for ("ok") |c| line.append(c);
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    try stdout.print("{s}\n", .{line.bytes[0..line.len]});
    try stdout.flush();
}

空の Line に append で文字 o と k を足し、たまった文字を表示するつもりのプログラムです。本来は ok と表示されるはずが、いまは Debug ビルドで実行すると何も表示せず、最初の append の self.bytes[self.len] で index out of bounds のパニックになります。

Answer

Wrong

var line: Line = undefined;

フィールドのデフォルト値 len: usize = 0 が使われるのは、構造体リテラルでそのフィールドを省いたときだけです。= undefined は構造体リテラルではないので、len も含めてすべてのフィールドが未定義の値になります。Debug ビルドでは未定義の値の領域が 0xaa のバイトで埋められるので、len は 12297829382473034410 になり、最初の self.bytes[self.len] で長さ8の配列の範囲を外れて止まります。1

Right

var line: Line = .{};

.{} はどのフィールドも指定しない構造体リテラルなので、すべてのフィールドにデフォルト値が入ります。len は0から始まり、append が bytes[0] と bytes[1] に書いて ok を表示します。bytes のデフォルト値は undefined のままですが、bytes[0..len] の外は読まないので問題になりません。

  1. langref は、undefined の値を使うことをバグとしています。0xaa で埋めるのは Debug と ReleaseSafe でバグを早く見つけるための実装上の振る舞いで、言語の意味としては保証されていません。ReleaseFast などでは len にどんな値が入っているかは決まらず、パニックせずに配列の外を書き換えることもあります。
