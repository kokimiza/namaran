snapshot.zig

const std = @import("std");

const Config = struct { port: u16 };

fn apply(cfg: *Config) void {
    cfg.port = 8080;
}

pub fn main(init: std.process.Init) !void {
    var cfg = Config{ .port = 80 };
    const before = &cfg;
    apply(&cfg);
    const changed = before.port != cfg.port;
    const stdout = std.Io.File.stdout();
    var w = stdout.writer(init.io, &.{});
    const out = &w.interface;
    try out.print("{}\n", .{changed});
}

このプログラムは、apply を呼ぶ前の設定を before に残しておき、呼んだあとの cfg と比べて、ポート番号が変わったかを表示するつもりのものです。apply はポートを 80 から 8080 に変えるので、本来は true と出力するはずが、いまは false と出力されます。

Answer

Wrong

const before = &cfg;
apply(&cfg);
const changed = before.port != cfg.port;

&cfg は cfg の中身の写しではなく、cfg が置かれている場所を指すポインタです。before は const ですが、const が止めるのは before に別のポインタを入れ直すことだけで、指す先の cfg が書き換わるのは止めません。apply(&cfg) がポートを 8080 にすると、before.port も同じ場所を読むので 8080 になります。8080 と 8080 を比べるので、changed はいつも false です。

Right

const before = cfg;
apply(&cfg);
const changed = before.port != cfg.port;

構造体の値を代入すると、Zig はその時点の中身を写します。const before = cfg; の before は cfg とは別の場所にある Config で、そのあと cfg が書き換わっても 80 のまま残ります。8080 と 80 を比べて true が出力されます。before.port の読み方は、ポインタでも値でも同じ . で書けるので1、& を1文字消すだけで直ります。

  1. Zig では、構造体へのポインタ p のフィールドを p.*.port と書かずに p.port と書けます。C の -> と . のような書き分けがないので、比べる行を読んでも before がポインタか値かは見分けられません。
