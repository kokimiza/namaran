jobs.zig

const std = @import("std");

const Job = struct {
    id: u32,
    notify: ?*const fn (u32) void = null,
};

fn log(id: u32) void {
    std.debug.print("done {d}\n", .{id});
}

pub fn main() void {
    const jobs = [_]Job{
        .{ .id = 1, .notify = &log },
        .{ .id = 2 },
    };
    for (jobs) |job| job.notify.?(job.id);
}

各ジョブには、終わったことを知らせる関数 notify を登録でき、登録したジョブだけにその関数を呼ぶつもりのプログラムです。main は log を登録したジョブ1と、何も登録していないジョブ2を順に処理するので、標準エラー出力に done 1 とだけ出して正常に終わるはずですが、いまは done 1 を出したあと attempt to use null value という panic で止まります。

Answer

Wrong

const Job = struct {
    id: u32,
    notify: ?*const fn (u32) void = null,
};

    for (jobs) |job| job.notify.?(job.id);

notify の型 ?*const fn (u32) void は、関数ポインタか null のどちらかを持つオプショナル型（optional）です。既定値が = null なので、ジョブ2のように .notify を書かずに作ると null が入ります。.? は「この値は null ではない」と決めつけて中身の関数ポインタを取り出す書き方で、null に使うと illegal behavior になります。

ジョブ1では notify に &log が入っているので、log(1) が呼ばれて done 1 が出ます。ジョブ2では job.notify.? が null を取り出そうとします。Debug ビルドはここに安全検査を入れているので、関数を呼ぶ前に attempt to use null value の panic で止まります。

Right

    for (jobs) |job| {
        if (job.notify) |f| f(job.id);
    }

if (job.notify) |f| は、notify が null でないときだけ中身の関数ポインタを f に取り出して本体を実行し、null なら何もしません。ジョブ1では f(1) が done 1 を出し、ジョブ2では呼び出しそのものが飛ばされるので、プログラムは正常に終わります。.? は、null になりえないことがほかの理由で分かっている値にだけ使います。
