labels.zig

const std = @import("std");

pub fn main(init: std.process.Init) !void {
    var out: [64]u8 = undefined;
    var w = std.Io.File.stdout().writer(init.io, &out);
    var buf: [16]u8 = undefined;
    var labels: [3][]const u8 = undefined;
    for (&labels, 0..) |*label, i| {
        label.* = try std.mem.print(&buf, "item{d}", .{i});
    }
    try w.interface.print("{s} {s} {s}\n", .{ labels[0], labels[1], labels[2] });
    try w.interface.flush();
}

item0、item1、item2 の3つのラベルを作って labels に取っておき、最後に1行で表示するつもりのプログラムです。本来は item0 item1 item2 が表示されるはずが、いまは item2 item2 item2 が表示されます。

Answer

Wrong

    var buf: [16]u8 = undefined;
    var labels: [3][]const u8 = undefined;
    for (&labels, 0..) |*label, i| {
        label.* = try std.mem.print(&buf, "item{d}", .{i});
    }

std.mem.print は渡された buf の先頭から結果を書き込み、書いた部分を指すスライス []u8 を返します。スライスは先頭を指すポインタと長さの組で、文字そのものは持っていません。3回の呼び出しはどれも同じ buf の先頭に書くので、labels の3つのスライスはすべて buf[0..5] を指します。ループが終わった時点で buf に残っているのは最後に書いた item2 だけで、3つのラベルはそれを3回読みます。

Right

    var bufs: [3][16]u8 = undefined;
    var labels: [3][]const u8 = undefined;
    for (&labels, 0..) |*label, i| {
        label.* = try std.mem.print(&bufs[i], "item{d}", .{i});
    }

ラベルごとに書き込み先を分けます。bufs は16バイトの配列を3つ並べた配列で、&bufs[i] は i 番目の配列を指す *[16]u8 です。これが []u8 に変換されて渡るので、i 番目のラベルは bufs[i] の中を指し、あとの呼び出しに上書きされません。labels はスライスを入れているだけなので、bufs は labels を使い終わるまで同じスコープに置いておく必要があります。
