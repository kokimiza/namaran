sum_unrolled.zig

const std = @import("std");

fn sum(xs: []const u32) u32 {
    var total: u32 = 0;
    var i: usize = 0;
    while (i < xs.len) : (i += 4) {
        inline for (0..4) |k| total += xs[i + k];
    }
    return total;
}

pub fn main(init: std.process.Init) !void {
    var buf: [64]u8 = undefined;
    var w = std.Io.File.stdout().writer(init.io, &buf);
    try w.interface.print("{d}\n", .{sum(&.{ 1, 2, 3, 4, 5, 6 })});
    try w.interface.flush();
}

sum は、ループの本体を inline for で4回分に展開し、スライス xs の値を4個ずつ足して合計を返すつもりの関数です。main は 1 から 6 までの6個を渡すので 21 が出力されるはずですが、いまは何も出力されず、index out of bounds: index 6, len 6 という panic で止まります。

Answer

Wrong

while (i < xs.len) : (i += 4) {
    inline for (0..4) |k| total += xs[i + k];
}

inline for (0..4) は、k が 0、1、2、3 の4文にコンパイル時に展開されます。何回足すかは実行時の xs.len を見ずに決まっていて、while の1周では必ず xs[i] から xs[i + 3] までを読みます。ところが while の条件 i < xs.len は、残りが1個でもあれば周に入ります。

6個のスライスでは、1周目に i = 0 で添字 0 から 3 を足し、2周目に i = 4 で周に入ります。残りは添字 4 と 5 の2個だけですが、展開された本体は添字 6 も読もうとします。Debug ビルドではスライスの添字が毎回検査されるので、ここで範囲外アクセス（illegal behavior）として panic になり、出力の前に止まります。

Right

while (i + 4 <= xs.len) : (i += 4) {
    inline for (0..4) |k| total += xs[i + k];
}
for (xs[i..]) |x| total += x;

while は、4個そろって残っているときだけ周に入るように i + 4 <= xs.len を条件にします。6個なら i = 0 の1周だけで終わり、i は 4 で止まります。4個に満たない端数の xs[4..] は、展開しない for で1個ずつ足すので、5 と 6 が加わって合計は 21 です。長さが4の倍数なら xs[i..] は空になり、端数のループは何もしません。
