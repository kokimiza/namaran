shelves.zig

const std = @import("std");

const Shelf = struct {
    stock: u32,
    fn absorb(self: *Shelf, other: *Shelf) void {
        self.stock += other.stock;
        other.stock = 0;
    }
};

pub fn main(init: std.process.Init) !void {
    var out: [64]u8 = undefined;
    var w = std.Io.File.stdout().writer(init.io, &out);
    var shelves = [_]Shelf{ .{ .stock = 3 }, .{ .stock = 4 }, .{ .stock = 5 } };
    for (&shelves) |*s| shelves[0].absorb(s);
    try w.interface.print("{d}\n", .{shelves[0].stock});
    try w.interface.flush();
}

3つの棚の在庫をすべて先頭の棚に集め、先頭の棚の在庫を表示するつもりのプログラムです。在庫は3個、4個、5個なので、本来は 12 と表示されるはずが、いまは 9 と表示されます。

Answer

Wrong

fn absorb(self: *Shelf, other: *Shelf) void {
    self.stock += other.stock;
    other.stock = 0;
}

shelves[0].absorb(s) は Shelf.absorb(&shelves[0], s) と同じ呼び出しで、self には先頭の棚のアドレスが入ります。ループは &shelves の全要素を回るので、1回目の s も先頭の棚 &shelves[0] です。このとき self と other は同じ棚を指します。

1回目では、まず self.stock += other.stock で在庫が3 + 3 = 6になり、続く other.stock = 0 が同じ棚を0にします。先頭の棚にあった3個はここで消えます。2回目と3回目は別の棚なので、4と5が正しく移って 0 + 4 + 5 = 9が表示されます。other を空にする処理も足す処理も、別々の棚なら正しく、同じ棚を指したときだけ順序が結果を壊します。

Right

fn absorb(self: *Shelf, other: *Shelf) void {
    const moved = other.stock;
    other.stock = 0;
    self.stock += moved;
}

移す量を先に moved に取り出し、other を空にしてから self に足します。別々の棚なら結果は前と変わりません。同じ棚なら、3を取り出して0にし、そこへ3を足し戻すので、在庫は3のまま残ります。2回目と3回目で4と5が足されて12になります。呼び出し側が自分自身を渡しても壊れないので、ループから先頭の棚を除く必要もありません。1

  1. 関数の先頭で if (self == other) return; と、同じ棚なら何もしないようにしても直ります。Zig のポインタは == でアドレスを比べられます。
