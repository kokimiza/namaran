trim_name.zig

const std = @import("std");

pub fn main(init: std.process.Init) !void {
    var buf: [64]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    const name = blk: {
        const raw = try init.gpa.dupe(u8, "  zig  ");
        defer init.gpa.free(raw);
        break :blk std.mem.trim(u8, raw, " ");
    };
    try stdout.print("[{s}]\n", .{name});
    try stdout.flush();
}

前後に空白の付いた名前を複製して受け取り、空白を除いた名前を name に入れて [名前] の形で表示するつもりのプログラムです。" zig " から本来は [zig] が出るはずが、いまは [UUU] のように、zig ではない3バイトが括弧の間に出ます。

Answer

Wrong

const name = blk: {
    const raw = try init.gpa.dupe(u8, "  zig  ");
    defer init.gpa.free(raw);
    break :blk std.mem.trim(u8, raw, " ");
};

defer に置いた文は、関数の終わりではなく、その defer を書いたブロックを抜けるときに実行されます。ここでのブロックは blk: のラベル付きブロックで、break :blk で値を持って抜ける瞬間に init.gpa.free(raw) が走ります。name に値が入ったときには、raw の7バイトはもう解放されています。

std.mem.trim は新しい文字列を作らず、渡されたスライスのうち前後の空白を除いた範囲を、同じメモリを指すスライスとして返します。name は raw の3バイト目から5バイト目を指したままなので、print は解放済みのメモリを読みます。解放した領域はアロケータが書き換えたり別の確保に回したりしてよい場所で、読んだ結果は決まりません。手元の Debug ビルドでは UUU が出ましたが、長さだけは name が覚えているので、括弧の間は3バイトになります。

Right

const name = blk: {
    const raw = try init.gpa.dupe(u8, "  zig  ");
    defer init.gpa.free(raw);
    break :blk try init.gpa.dupe(u8, std.mem.trim(u8, raw, " "));
};
defer init.gpa.free(name);

trim の結果を init.gpa.dupe で新しく確保したメモリへコピーしてから、ブロックの値にします。ブロックを抜けるときに raw が解放されても、name は raw とは別のメモリを指しているので、zig の3文字が残ります。今度は name の側を解放しなければならないので、ブロックの外、name を宣言した直後に defer init.gpa.free(name); を置きます。この defer は main の本体というブロックに属するので、print と flush が終わって main を抜けるときに実行されます。1

  1. ラベル付きブロックでなくても同じです。if や for の本体の { } も1つのブロックで、その中の defer は本体を抜けるたびに実行されます。for の本体なら、周回ごとに走ります。
