names.zig

const std = @import("std");

pub fn main(init: std.process.Init) !void {
    var out: [64]u8 = undefined;
    var w = std.Io.File.stdout().writer(init.io, &out);
    const lines = [_][]const u8{ "alice,3", "bo,5" };
    var buf: [16]u8 = undefined;
    var names: [2][]const u8 = undefined;
    for (lines, 0..) |line, i| {
        @memcpy(buf[0..line.len], line);
        var it = std.mem.splitScalar(u8, buf[0..line.len], ',');
        names[i] = it.first();
    }
    try w.interface.print("{s} {s}\n", .{ names[0], names[1] });
    try w.interface.flush();
}

名前,点数 の形の行を1行ずつ作業用の buf に読み込み(ここでは @memcpy で読み込みの代わりにしています)、各行の名前を names にためてから並べて表示するつもりのプログラムです。行が alice,3 と bo,5 のとき、いまは bo,5e bo が表示されます。本来は alice bo です。

Answer

Wrong

var buf: [16]u8 = undefined;
var names: [2][]const u8 = undefined;
for (lines, 0..) |line, i| {
    @memcpy(buf[0..line.len], line);
    var it = std.mem.splitScalar(u8, buf[0..line.len], ',');

splitScalar は文字列をコピーせず、渡されたスライスの中の範囲を指すスライスを返します。1行目で it.first() が返すのは「buf の先頭から5バイト」で、names[0] に入るのも、文字の alice ではなく、この位置と長さの組です。2行目を読み込むと、buf の先頭4バイトが bo,5 で上書きされ、5バイト目には1行目の alice,3 の e が残ります。names[0] は同じ5バイトを指したままなので、表示すると bo,5e になります。

names[1] は2行目の先頭2バイトを指す正しい値に見えますが、これも buf を指していて、3行目があれば同じように書き換わります。alice だけが消えて bo になるのではなく、文字は2行目のもの、長さは1行目のものという組み合わせになるのは、スライスが位置と長さだけを持ち、中身は buf から読むからです。

Right

var bufs: [2][16]u8 = undefined;
var names: [2][]const u8 = undefined;
for (lines, 0..) |line, i| {
    @memcpy(bufs[i][0..line.len], line);
    var it = std.mem.splitScalar(u8, bufs[i][0..line.len], ',');

行ごとに別の領域 bufs[i] に読み込めば、names[i] が指す bufs[i] を後の行が上書きすることはなく、alice と bo がそのまま残ります。行数が前もって分からないときは、names に入れる前に、アロケータで確保した領域へ名前をコピーします。split や tokenize で取り出したスライスは、元のバッファが書き換わらない間だけ有効だと考えて扱います。
