stats.zig

const std = @import("std");

const Stats = struct {
    count: u32 = 0,
    total: u32 = 0,
};

pub fn main(init: std.process.Init) !void {
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    var stats: Stats = undefined;
    for ([_]u32{ 4, 7, 1 }) |x| {
        stats.count += 1;
        stats.total += x;
    }
    try stdout.print("{d} {d}\n", .{ stats.count, stats.total });
    try stdout.flush();
}

配列 { 4, 7, 1 } の要素の個数と合計を、フィールドの既定値を0にした Stats に数えて出力するプログラムです。本来は 3 12 と出力されるはずが、いまは大きなでたらめの数が出ます。手元の Debug ビルドでは 2863311533 2863311542 と出ましたが、値はビルドの設定によって変わります。

Answer

Wrong

var stats: Stats = undefined;

undefined で初期化した変数は、中身が決まらないまま置かれるだけで、フィールドの既定値は使われません。既定値が入るのは、構造体リテラルでそのフィールドを書かなかったときです。そのため stats.count と stats.total は決まらない値から数え始め、そこに 1 や要素が足されていきます。Debug ビルドでは、undefined の変数のバイトがすべて 0xaa で埋められるので、u32 の初期値は 0xaaaaaaaa（10進数で 2863311530）になり、それに3と12を足した数が出力されます1。

Right

var stats: Stats = .{};

.{} はフィールドを1つも書かない構造体リテラルなので、count と total はどちらも宣言の既定値 0 から始まります。ループで count に1が3回、total に4、7、1が足され、3 12 が出力されます。

  1. 0xaa で埋めるのは、決まらない値を使ったときに気付きやすくするための Debug ビルドの振る舞いで、言語として保証された値ではありません。ReleaseFast などの最適化したビルドでは埋める処理が行われず、出てくる値は決まりません。
