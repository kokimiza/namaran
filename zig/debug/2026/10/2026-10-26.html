job_log.zig

const std = @import("std");

const Job = struct { id: u32 = 0, name: []const u8 };
var next_id: u32 = 42;

fn start(job: *Job, out: *std.Io.Writer) !void {
    errdefer out.print("job {d} rolled back\n", .{job.id}) catch {};
    job.id = next_id;
    next_id += 1;
    errdefer job.id = 0;
    if (job.name.len == 0) return error.EmptyName;
}

pub fn main(init: std.process.Init) !void {
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const out = &stdout_writer.interface;
    var job: Job = .{ .name = "" };
    start(&job, out) catch |err| try out.print("error: {s}\n", .{@errorName(err)});
    try out.flush();
}

start は、ジョブにIDを割り当てて開始する関数です。失敗したときはIDを 0 に戻し、どのIDのジョブを巻き戻したかを out に書くつもりです。発生したエラーは main が catch |err| で受け取って書きます。名前が空のジョブを渡すと、本来の出力は job 42 rolled back と error: EmptyName の2行ですが、いまは1行目が job 0 rolled back になります。

Answer

Wrong

    errdefer out.print("job {d} rolled back\n", .{job.id}) catch {};
    job.id = next_id;
    next_id += 1;
    errdefer job.id = 0;

errdefer は、関数がエラーで終わるときに、置いたのとは逆の順に実行されます。return error.EmptyName で抜けると、あとに置いた errdefer job.id = 0 が先に実行され、ログの errdefer はそのあとです。errdefer の式は置いた時点ではなく実行される時点で評価されるので、ログが job.id を読むときには、すでに 0 に戻っています。

ログの errdefer を job.id = next_id より前に置いたので、割り当てる前の 0 が記録された、と読みたくなります。しかしログの行を割り当ての直後に移しても、errdefer job.id = 0 より前にある限り実行はそれより後なので、出力は変わりません。

Right

    job.id = next_id;
    next_id += 1;
    errdefer job.id = 0;
    errdefer out.print("job {d} rolled back\n", .{job.id}) catch {};

ログの errdefer を、IDを戻す errdefer よりあとに置きます。逆の順に実行されるので、ログが先に書かれ、そのときの job.id はまだ 42 です。errdefer は、巻き戻す状態を作った行の直後に作った順で置くと、巻き戻しが作ったときと逆の順になり、どの後始末もまだ巻き戻されていない状態を読めます。エラーの名前は、errdefer からは参照できないので、main の catch |err| で書いています。1

  1. 以前の版の Zig には、errdefer |err| と書いて返すエラーを受け取る構文がありました。Zig 0.17 の文法では errdefer の後ろに置けるのは式だけで、この構文はコンパイルエラーになります。
