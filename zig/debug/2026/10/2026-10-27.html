sum_config.zig

const std = @import("std");

fn sumValues(lines: []const []const u8) !u32 {
    var total: u32 = 0;
    var i: usize = 0;
    while (i < lines.len) {
        if (std.mem.startsWith(u8, lines[i], "#")) continue;
        total += try std.fmt.parseInt(u32, lines[i], 10);
        i += 1;
    }
    return total;
}

pub fn main(init: std.process.Init) !void {
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    try stdout.print("{d}\n", .{try sumValues(&.{ "10", "# skip", "32" })});
    try stdout.flush();
}

設定の各行に書かれた数を合計し、# で始まる注釈の行は飛ばすつもりの sumValues です。main は "10"、"# skip"、"32" の3行を渡します。本来は 42 と出力されるはずが、いまは何も出力されないまま、プログラムが終わらなくなります。

Answer

Wrong

    while (i < lines.len) {
        if (std.mem.startsWith(u8, lines[i], "#")) continue;
        total += try std.fmt.parseInt(u32, lines[i], 10);
        i += 1;
    }

i を進める i += 1 は本体の最後にあり、continue はそこを通らずに次の回へ戻ります。2行目の "# skip" で continue すると i は1のまま変わらず、次の回もまた同じ注釈の行を見て continue します。条件 i < lines.len はずっと真なので、ループは終わりません。

Right

    while (i < lines.len) : (i += 1) {
        if (std.mem.startsWith(u8, lines[i], "#")) continue;
        total += try std.fmt.parseInt(u32, lines[i], 10);
    }

i += 1 を本体から while の継続式(continue expression): (i += 1) へ移します。継続式は本体を最後まで実行したときにも、continue で次の回へ進むときにも実行されるので、注釈の行を飛ばしても i は進みます。本体にあとから continue を足しても、添字の更新を通り過ぎることはありません。
