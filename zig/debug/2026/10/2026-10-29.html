count_odd.zig

const std = @import("std");

fn countOdd(xs: []const i32) usize {
    var n: usize = 0;
    for (xs) |x| {
        if (@rem(x, 2) == 1) n += 1;
    }
    return n;
}

pub fn main(init: std.process.Init) !void {
    var buf: [256]u8 = undefined;
    var stdout_writer = std.Io.File.stdout().writer(init.io, &buf);
    const stdout = &stdout_writer.interface;
    try stdout.print("{d}\n", .{countOdd(&.{ 3, -5, 8, -1, 7 })});
    try stdout.flush();
}

スライスの中の奇数の個数を返す countOdd です。main は 3, -5, 8, -1, 7 を渡します。奇数は3、-5、-1、7の4つなので本来は 4 と出力されるはずが、いまは 2 と出力されます。

Answer

Wrong

        if (@rem(x, 2) == 1) n += 1;

@rem は商を0の側へ切り捨てる割り算の余りで、余りは割られる数と同じ符号になります。-5は 2 * (-2) + (-1) なので、@rem(-5, 2) は -1、@rem(-1, 2) も -1 です。負の奇数では == 1 が偽になり、数えられるのは正の奇数の3と7だけで、countOdd は2を返します。

Right

        if (@mod(x, 2) == 1) n += 1;

@mod は商を負の側へ切り捨てる割り算の余りで、割る数が正なら余りは0以上になります。-5は 2 * (-3) + 1 なので @mod(-5, 2) は1で、奇数ならどの符号でも余りは1になり、4つとも数えられます。比べる側を @rem(x, 2) != 0 に変えても直ります1。

  1. C や Java の % も @rem と同じく割られる数の符号を持つので、x % 2 == 1 は同じように負の奇数を取りこぼします。Zig は符号付き整数に % を使うことを許さず、実行時に値が決まる場合は @rem か @mod のどちらかを書かせます。
