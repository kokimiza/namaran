dupe_all.zig

const Diag = struct { copied: usize = 0, err: ?anyerror = null };

fn dupeAll(gpa: std.mem.Allocator, items: []const []const u8, diag: *Diag) ![][]u8 {
}

文字列のスライス items の各要素を gpa で確保した領域へ複製し、複製した文字列のスライスを返す dupeAll を書いてください。返すスライスとそれぞれの文字列は、呼び出し側が解放します。確保に1回でも失敗したら、それまでに確保した領域をすべて解放してからそのエラーを返し、diag.copied に複製を終えていた文字列の数を、diag.err に返したエラーを書きます。成功したときの diag は、copied が items.len で err が null です。items が空なら空のスライスを返します。入力例 3 の failing は、2回の確保に成功したあと3回目から失敗するアロケータです。

入力例 1

var diag: Diag = .{};
dupeAll(gpa, &.{ "ab", "cd" }, &diag)

出力例 1

.{ "ab", "cd" }
diag == .{ .copied = 2, .err = null }

入力例 2

dupeAll(gpa, &.{}, &diag)

出力例 2

.{}
diag == .{ .copied = 0, .err = null }

入力例 3

var failing = std.testing.FailingAllocator.init(gpa, .{ .fail_index = 2 });
dupeAll(failing.allocator(), &.{ "ab", "cd", "ef" }, &diag)

出力例 3

error.OutOfMemory
diag == .{ .copied = 1, .err = error.OutOfMemory }

fn dupeAll(gpa: std.mem.Allocator, items: []const []const u8, diag: *Diag) ![][]u8 {
    diag.* = .{};
    return dupeEach(gpa, items, &diag.copied) catch |err| {
        diag.err = err;
        return err;
    };
}

fn dupeEach(gpa: std.mem.Allocator, items: []const []const u8, n: *usize) ![][]u8 {
    const out = try gpa.alloc([]u8, items.len);
    errdefer gpa.free(out);
    errdefer for (out[0..n.*]) |s| gpa.free(s);
    while (n.* < items.len) : (n.* += 1) out[n.*] = try gpa.dupe(u8, items[n.*]);
    return out;
}

後始末は dupeEach の errdefer が受け持ちます。errdefer は関数がエラーで終わるときに、置いたのとは逆の順に実行されるので、先に out[0..n.*] の文字列を解放し、そのあとで out 自体を解放します。解放したあとの out を読むことはありません。n.* は errdefer を置いた時点ではなく実行される時点で読まれるので、失敗したときには複製を終えた数になっています。out の確保そのものに失敗した場合は、どちらの errdefer にもまだ届いていないので、何も解放しません。

文字列を解放する errdefer を繰り返しの本体に置くと、本体を抜けるたびにその有効範囲が終わるので、後の回で失敗したときに前の回の文字列を解放できません。そのため数える変数を繰り返しの外に置き、関数の終わりで一度に解放しています。

Zig 0.17 の errdefer は、返そうとしているエラーを受け取れません。1 そこで dupeAll は、後始末を済ませて戻ってきた dupeEach のエラーを catch |err| で受け取り、diag.err に書いてからそのまま返します。数は &diag.copied を渡して dupeEach に直接数えさせるので、失敗した時点の値がそのまま残ります。

検証では std.testing.FailingAllocator で1回目から4回目までの確保をそれぞれ失敗させ、diag の値と、確保したバイト数と解放したバイト数が一致することを確かめました。掲載したのは参考実装で、唯一の正解ではありません。1つの関数の中で、ラベル付きブロックの結果を catch で受ける書き方もできます。

  1. 以前の版の Zig には、errdefer |err| と書いて返すエラーを受け取る構文がありました。Zig 0.17 の文法では errdefer の後ろに置けるのは式だけで、この構文はコンパイルエラーになります。
