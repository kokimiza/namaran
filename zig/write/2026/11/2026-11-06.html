stack.zig

fn Stack(comptime T: type, comptime cap: usize) type {
}

要素の型 T と容量 cap を受け取り、最大 cap 個の要素を積める固定長のスタックの型を返す Stack を書いてください。返す型には3つの関数を持たせます。init() は空のスタックを返します。push(x) は x を積んで true を返し、すでに cap 個あれば何もせず false を返します。pop() は最後に積んだ要素を取り除いて返し、空なら null を返します。メモリは確保しません。例では、var s = Stack(u8, 2).init(); と作った s に対して呼び出しを順に行い、最後の呼び出しの結果を書きます。

入力例 1

_ = s.push(1); _ = s.push(2); s.push(3)

出力例 1

false

入力例 2

_ = s.push(1); _ = s.push(2); s.pop()

出力例 2

2

入力例 3

s.pop()

出力例 3

null

fn Stack(comptime T: type, comptime cap: usize) type {
    return struct {
        const Self = @This();
        items: [cap]T = undefined,
        len: usize = 0,

        fn init() Self {
            return .{};
        }

        fn push(self: *Self, x: T) bool {
            if (self.len == cap) return false;
            self.items[self.len] = x;
            self.len += 1;
            return true;
        }

        fn pop(self: *Self) ?T {
            if (self.len == 0) return null;
            self.len -= 1;
            return self.items[self.len];
        }
    };
}

Stack は名前の無い struct を返し、その構造体が型であると同時に、init・push・pop を置く名前空間になります。名前が無いので、メソッドの引数や戻り値に自分の型を書くには @This() を使い、const Self = @This(); で名前を付けます。フィールドは cap 個分の配列 items と、積んだ個数 len だけです。cap が comptime 引数なので配列の長さに使え、メモリを確保しなくて済みます。items と len に既定値を付けたので、init は .{} を返すだけです。これは参考実装で、唯一の正解ではありません。init の代わりに pub const empty: Self = .{}; を置き、var s: Stack(u8, 2) = .empty; と書かせる形もあります。

init は self を受け取らないので、Stack(u8, 2).init() のように型の名前空間から呼びます。push と pop は s 自身の len を変えるので、self を *Self で受けます。s.push(3) と書けば、Zig が s のアドレスを渡します。self: Self と値で受けると self は書き換えられない引数になり、self.len += 1 がコンパイルエラーになります。pop の戻り値は ?T にしたので、空のときは null を、そうでなければ T の値をそのまま返せます。入力例1は2個積んだ時点で len == cap なので3回目の push は false、入力例2は最後に積んだ 2 が返り、入力例3は len が0なので null です。
