min_max.zig

const MinMax = struct {
    min: ?i32 = null,
    max: ?i32 = null,
};

fn minMax(xs: []const i32) MinMax {
}

xs の最小値を min に、最大値を max に入れた MinMax を返す minMax を書いてください。xs が空なら、min も max も null のままの MinMax を返します。例では、返した値を .{ .min = 最小値, .max = 最大値 } の形で書きます。

入力例 1

minMax(&.{ 3, -2, 7 })

出力例 1

.{ .min = -2, .max = 7 }

入力例 2

minMax(&.{5})

出力例 2

.{ .min = 5, .max = 5 }

入力例 3

minMax(&.{})

出力例 3

.{ .min = null, .max = null }
Reference 
fn minMax(xs: []const i32) MinMax {
    var r: MinMax = .{};
    for (xs) |x| {
        r.min = @min(r.min orelse x, x);
        r.max = @max(r.max orelse x, x);
    }
    return r;
}

var r: MinMax = .{}; は、どのフィールドも指定しない構造体リテラルなので、min と max にはフィールドのデフォルト値 null が入ります。この null が「まだ1つも見ていない」ことを表すので、xs が空ならループを1度も回らず、そのまま入力例3の値を返せます。

ループの中の r.min orelse x は、r.min が null なら x を、そうでなければ中の値を返します。最初の要素では @min(x, x) になって x がそのまま入り、2つ目からは今までの最小値と x の小さいほうが入ります。@min の結果は i32 で、?i32 のフィールドにはそのまま代入できます。

掲載したのは参考実装で、唯一の正解ではありません。if (r.min) |m| で null かどうかを分けてから比べてもかまいません。
