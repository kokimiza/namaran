join_ints.zig

const std = @import("std");

fn joinInts(gpa: std.mem.Allocator, xs: []const i64) ![]u8 {
}

整数の並び xs を10進数で書き、間に ", " を挟んでつないだ文字列を返す joinInts を書いてください。文字列は std.Io.Writer.Allocating に書いて作り、gpa で確保した領域として返します。返した領域は、呼び出し側が gpa.free で解放します。xs が空なら空の文字列を返します。途中で確保に失敗してエラーを返すときは、それまでに確保した領域をすべて解放してから返します。

入力例 1

joinInts(gpa, &.{ 1, 2, 3 })

出力例 1

"1, 2, 3"

入力例 2

joinInts(gpa, &.{-5})

出力例 2

"-5"

入力例 3

joinInts(gpa, &.{})

出力例 3

""

fn joinInts(gpa: std.mem.Allocator, xs: []const i64) ![]u8 {
    var aw: std.Io.Writer.Allocating = .init(gpa);
    defer aw.deinit();
    for (xs, 0..) |x, i| {
        if (i > 0) try aw.writer.writeAll(", ");
        try aw.writer.print("{d}", .{x});
    }
    return aw.toOwnedSlice();
}

std.Io.Writer.Allocating は、gpa で確保した領域をそのまま Writer のバッファにし、足りなくなると確保し直して広げます。書いた文字はバッファに残り、それが結果になるので、ファイルへ書くときのような flush は要りません。区切りは2つ目以降の要素の前にだけ書くので、要素が1つなら -5 だけ、空なら何も書かずに空の文字列になります。

返し方は、defer aw.deinit() と toOwnedSlice の組み合わせで決まります。deinit は aw が持っている領域を解放します。defer で書いておけば、try が確保の失敗でエラーを返す道でも解放され、問題文の条件を満たします。ただし defer は、成功して return するときも、return の式を評価したあとに実行されます。return aw.written(); と書くと、返したスライスの指す領域を deinit がその場で解放してしまい、呼び出し側は解放済みの領域を読み、gpa.free で二重に解放します。

toOwnedSlice は、書いた部分を gpa で確保したスライスとして取り出し、aw を何も持たない状態に戻します。その後の deinit は解放するものがないので何もせず、スライスの持ち主は呼び出し側に移ります。toOwnedSlice も確保し直しで失敗することがあり、戻り値の型 ![]u8 はそのエラーと、print が返す error.WriteFailed をまとめて推論させています。

掲載したのは参考実装で、唯一の正解ではありません。defer の代わりに errdefer aw.deinit(); と書いても、toOwnedSlice が成功したあとは解放するものが残らないので、同じように動きます。
