run_start.zig

fn start(s: []const u8) ?u64 {
}

s は数字だけからなる文字列です。s が、0 以上の整数 k から始まる連続した2個以上の整数 k, k + 1, k + 2, … を、先頭に余分な 0 を付けない10進表記で区切りなしにつないだものなら k を返し、どんな k でもそうならないなら null を返す start を書いてください。そのような k が複数あるときは、桁数がいちばん少ないものを返します。s の長さは 20 文字以下とします。

入力例 1

start("9101112")

出力例 1

9

入力例 2

start("99100")

出力例 2

99

入力例 3

start("0102")

出力例 3

null

fn matches(s: []const u8, k: u64) bool {
    var buf: [20]u8 = undefined;
    var rest = s;
    var n = k;
    while (rest.len > 0) : (n += 1) {
        const t = std.fmt.bufPrint(
            &buf,
            "{d}",
            .{n},
        ) catch return false;
        if (!std.mem.startsWith(u8, rest, t))
            return false;
        rest = rest[t.len..];
    }
    return true;
}

fn start(s: []const u8) ?u64 {
    for (1..s.len) |len| {
        const k = std.fmt.parseInt(
            u64,
            s[0..len],
            10,
        ) catch continue;
        if (matches(s, k)) return k;
    }
    return null;
}

s を先頭から数に切り分けようとしても、数の桁数は 9 から 10、99 から 100 のように途中で増えるので、どこで切ればよいかは s だけからは決まりません。ところが最初の数 k さえ決まれば、続く数は k + 1, k + 2, … と決まり、その10進表記も決まります。参考実装はこの向きで解きます。start が最初の数の桁数 len を短いほうから順に仮に決め、s[0..len] を std.fmt.parseInt で k に読み、matches が k から順に std.fmt.bufPrint で書き出した文字列を s の先頭から照合していきます。len を s.len 未満に限っているので、最初の数のあとに少なくとも1個の数が続きます。bufPrint の書き出し先は、u64 の最大値の20桁が入る大きさです。掲載したのは参考実装で、唯一の正解ではありません。

入力例 1 では、len が 1 のとき k = 9 で、9、10、11、12 と書き出した文字列が s を残らず覆うので 9 を返します。入力例 2 は len が 1 の 9 では次の 10 が合わず、len が 2 の 99 で 100 まで合います。入力例 3 が null になるのは、照合を書き出した側の文字列で行っているからです。s[0..2] の "01" も parseInt は 1 と読みますが、1 を書き出すと "1" で、s の先頭の "0" と合いません。parseInt で読み進めて「次の数が1大きいか」だけを確かめる書き方だと、01 と 02 を受け入れて 1 を返してしまいます。読むのと書くのは対になる処理ですが、先頭の 0 のように、読むときには捨てられて書くときには作られない情報があるので、照合は書き出した表記どうしで行います。
