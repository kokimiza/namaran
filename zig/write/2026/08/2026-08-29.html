load_config.zig

const Server = struct {
    port: u16 = 8080,
    workers: u8 = 4,
};

fn load(comptime T: type, pairs: []const [2][]const u8) !T {
}

全フィールドが整数型で、どのフィールドにも既定値がある構造体型 T について、設定を読み込む load を書いてください。pairs の各要素は、フィールド名と10進数の文字列の組です。すべてのフィールドが既定値の状態から始め、組を先頭から順に見て、名前の合うフィールドをその数に書き換えた値を返します。同じ名前が何度も出たら後のものが残り、pairs が空なら既定値のままです。どのフィールドにも合わない名前があれば error.UnknownKey を返し、数として読めない文字列やフィールドの型に収まらない数なら、std.fmt.parseInt の返すエラーをそのまま返します。

入力例 1

load(Server, &.{.{ "port", "9000" }})

出力例 1

.{ .port = 9000, .workers = 4 }

入力例 2

load(Server, &.{ .{ "workers", "8" }, .{ "workers", "2" } })

出力例 2

.{ .port = 8080, .workers = 2 }

入力例 3

load(Server, &.{.{ "host", "1" }})

出力例 3

error.UnknownKey
Reference 
fn load(comptime T: type, pairs: []const [2][]const u8) !T {
    var result: T = .{};
    const info = @typeInfo(T).@"struct";
    for (pairs) |pair| {
        inline for (info.field_names, info.field_types) |name, F| {
            if (std.mem.eql(u8, pair[0], name)) {
                @field(result, name) = try std.fmt.parseInt(F, pair[1], 10);
                break;
            }
        } else return error.UnknownKey;
    }
    return result;
}

出発点の .{} は、フィールドを1つも書かない構造体リテラルで、すべてのフィールドが宣言の既定値になります。既定値の無いフィールドが1つでもあると、ここでコンパイルエラーになります。var result: T = undefined; から始めると、既定値は入らず、書き換えなかったフィールドの中身は決まらないままです。

@typeInfo(T).@"struct" は、フィールドの名前の並び field_names と型の並び field_types をコンパイル時に持っています。@field(result, name) の name はコンパイル時に決まっていなければならず、フィールドごとに型 F も違うので、この2つの並びは普通の for では回せません。inline for はループ本体をフィールドの数だけ展開し、展開したそれぞれで name と F が定数になります。比べる相手の pair[0] は実行時の文字列なので、比較そのものは実行時に std.mem.eql で行います。

名前が合えば、std.fmt.parseInt(F, …) がそのフィールドの型で文字列を読みます。Server の workers は u8 なので、"300" を渡すと error.Overflow になり、try がそれを呼び出し元へ返します。書き換えたら break で残りのフィールドを飛ばし、どのフィールドとも合わずにループが最後まで回ったときだけ、else の return error.UnknownKey に進みます。

掲載したのは参考実装で、唯一の正解ではありません。inline for の else を使わず、合ったかどうかを bool の変数に記録して、ループのあとで調べる書き方もあります。
